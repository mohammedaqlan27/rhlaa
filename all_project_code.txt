import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:flutter/foundation.dart';
import '../models/agency.dart';
import '../models/agency_transaction.dart';

class AgenciesService {
  final FirebaseFirestore _firestore;

  AgenciesService({FirebaseFirestore? firestore})
      : _firestore = firestore ?? FirebaseFirestore.instance;

  CollectionReference get _col => _firestore.collection('agencies');
  CollectionReference get _transactionsCol => _firestore.collection('agency_transactions');

  // Helper محلي لتحويل القيم إلى double بأمان (يحل محل استدعاءات Agency.toDoubleSafe إن لم تكن متاحة)
  double _toDoubleSafe(dynamic v) {
    if (v == null) return 0.0;
    if (v is double) return v;
    if (v is int) return v.toDouble();
    if (v is num) return v.toDouble();
    if (v is String) return double.tryParse(v) ?? 0.0;
    return 0.0;
  }

  /// إنشاء مكتب جديد
  Future<String> createAgency({
    required String agencyName,
    required String managerName,
    required dynamic location,
    required String mobileNumber,
    double initialBalance = 0.0,
    bool isActive = true,
  }) async {
    final now = Timestamp.now();
    final docRef = await _col.add({
      'agencyName': agencyName,
      'managerName': managerName,
      'location': location,
      'mobileNumber': mobileNumber,
      'walletBalance': initialBalance,
      'isActive': isActive,
      'createdAt': now,
      'updatedAt': now,
    });
    return docRef.id;
  }

  /// جلب مكتب حسب المعرف
  Future<Agency?> getAgencyById(String id) async {
    try {
      final doc = await _col.doc(id).get();
      if (!doc.exists) return null;
      return Agency.fromDoc(doc);
    } catch (e) {
      debugPrint('AgenciesService.getAgencyById error: $e');
      rethrow;
    }
  }

  /// جلب كل المكاتب
  Future<List<Agency>> fetchAllAgencies({bool onlyActive = false}) async {
    try {
      Query q = _col.orderBy('createdAt', descending: true);
      if (onlyActive) q = q.where('isActive', isEqualTo: true);
      final snap = await q.get();
      return snap.docs.map((d) => Agency.fromDoc(d)).toList();
    } catch (e) {
      debugPrint('AgenciesService.fetchAllAgencies error: $e');
      return <Agency>[];
    }
  }

  /// تحديث بيانات المكتب (جزئي)
  Future<void> updateAgency(String id, Map<String, dynamic> updates) async {
    try {
      updates.removeWhere((key, value) => value == null);
      updates['updatedAt'] = Timestamp.now();
      await _col.doc(id).update(updates);
    } catch (e) {
      debugPrint('AgenciesService.updateAgency error: $e');
      rethrow;
    }
  }

  /// حذف المكتب
  Future<void> deleteAgency(String id) async {
    try {
      await _col.doc(id).delete();
    } catch (e) {
      debugPrint('AgenciesService.deleteAgency error: $e');
      rethrow;
    }
  }

  /// تفعيل/تعطيل المكتب
  Future<void> setAgencyActive(String id, bool isActive) async {
    try {
      await _col.doc(id).update({'isActive': isActive, 'updatedAt': Timestamp.now()});
    } catch (e) {
      debugPrint('AgenciesService.setAgencyActive error: $e');
      rethrow;
    }
  }

  /// شحن رصيد المحفظة للمكتب (transaction آمن) مع تسجيل المعاملة
  Future<void> creditWallet(String agencyId, double amount, {String? reason, String? performedBy}) async {
    if (amount <= 0) return;
    final docRef = _col.doc(agencyId);
    await _firestore.runTransaction((tx) async {
      final snapshot = await tx.get(docRef);
      if (!snapshot.exists) throw Exception('Agency not found');
      final current = (snapshot.data() as Map<String, dynamic>?)?['walletBalance'];
      final currentBalance = _toDoubleSafe(current);
      final newBalance = currentBalance + amount;
      tx.update(docRef, {
        'walletBalance': newBalance,
        'updatedAt': Timestamp.now(),
      });

      // سجل المعاملة
      final txDoc = _transactionsCol.doc();
      final now = Timestamp.now();
      tx.set(
        txDoc,
        AgencyTransaction(
          id: txDoc.id,
          agencyId: agencyId,
          amount: amount,
          type: 'credit',
          reason: reason ?? 'manual_credit',
          performedBy: performedBy,
          createdAt: now,
        ).toMap(),
      );
    });
  }

  /// خصم من رصيد المحفظة للمكتب (transaction آمن) مع تسجيل المعاملة
  Future<void> debitWallet(String agencyId, double amount, {String? reason, String? performedBy}) async {
    if (amount <= 0) return;
    final docRef = _col.doc(agencyId);
    await _firestore.runTransaction((tx) async {
      final snapshot = await tx.get(docRef);
      if (!snapshot.exists) throw Exception('Agency not found');
      final current = (snapshot.data() as Map<String, dynamic>?)?['walletBalance'];
      final currentBalance = _toDoubleSafe(current);
      if (currentBalance < amount) throw Exception('Insufficient agency wallet balance');
      final newBalance = currentBalance - amount;
      tx.update(docRef, {
        'walletBalance': newBalance,
        'updatedAt': Timestamp.now(),
      });

      // سجل المعاملة
      final txDoc = _transactionsCol.doc();
      final now = Timestamp.now();
      tx.set(
        txDoc,
        AgencyTransaction(
          id: txDoc.id,
          agencyId: agencyId,
          amount: amount,
          type: 'debit',
          reason: reason ?? 'manual_debit',
          performedBy: performedBy,
          createdAt: now,
        ).toMap(),
      );
    });
  }

  /// جلب سجل المعاملات لمكتب معين (مرتّب تنازلياً)
  Future<List<AgencyTransaction>> fetchTransactions(String agencyId, {int limit = 100}) async {
    try {
      final snap = await _transactionsCol
          .where('agencyId', isEqualTo: agencyId)
          .orderBy('createdAt', descending: true)
          .limit(limit)
          .get();
      return snap.docs.map((d) => AgencyTransaction.fromDoc(d)).toList();
    } catch (e) {
      debugPrint('AgenciesService.fetchTransactions error: $e');
      return <AgencyTransaction>[];
    }
  }
}

// lib/services/auth_service.dart
import 'package:firebase_auth/firebase_auth.dart';
import 'package:cloud_firestore/cloud_firestore.dart';
import '../models/app_user.dart';
import 'package:flutter/foundation.dart';

class AuthService {
  final FirebaseAuth _auth;
  final FirebaseFirestore _firestore;

  // الآن الكونستركتور يقبل تبعيات اختيارية (للاختبارات أو المحاكيات)
  AuthService({FirebaseAuth? auth, FirebaseFirestore? firestore})
      : _auth = auth ?? FirebaseAuth.instance,
        _firestore = firestore ?? FirebaseFirestore.instance;

  // إذا أردت تفعيل محاكيات في وقت التشغيل (اختياري)
  void useEmulators({required String host, required int authPort, required int firestorePort}) {
    try {
      _auth.useAuthEmulator(host, authPort);
    } catch (_) {}
    try {
      _firestore.useFirestoreEmulator(host, firestorePort);
    } catch (_) {}
  }

  Stream<User?> get authStateChanges => _auth.authStateChanges();

  String _handleAuthError(FirebaseAuthException e) {
    switch (e.code) {
      case 'user-not-found':
        return 'هذا البريد الإلكتروني غير مسجل.';
      case 'wrong-password':
        return 'كلمة المرور غير صحيحة.';
      case 'email-already-in-use':
        return 'هذا البريد الإلكتروني مستخدم بالفعل.';
      case 'weak-password':
        return 'كلمة المرور ضعيفة جداً.';
      case 'invalid-email':
        return 'صيغة البريد الإلكتروني غير صحيحة.';
      case 'network-request-failed':
        return 'تأكد من اتصالك بالإنترنت.';
      case 'too-many-requests':
        return 'تم حظر المحاولات مؤقتاً. حاول لاحقاً.';
      case 'operation-not-allowed':
        return 'طريقة المصادقة غير مفعلة في إعدادات Firebase.';
      default:
        return e.message ?? 'حدث خطأ أثناء المصادقة.';
    }
  }

  Future<User?> signIn(String email, String password) async {
    try {
      final result = await _auth.signInWithEmailAndPassword(email: email.trim(), password: password);
      return result.user;
    } on FirebaseAuthException catch (e) {
      throw _handleAuthError(e);
    } catch (e) {
      throw 'حدث خطأ غير متوقع أثناء تسجيل الدخول.';
    }
  }

  Future<User?> register({
    required String name,
    required String nickname,
    required String email,
    required String password,
    required String role,
  }) async {
    try {
      final result = await _auth.createUserWithEmailAndPassword(email: email.trim(), password: password);
      final user = result.user;
      if (user != null) {
        await _createUserRecord(user.uid, name, nickname, email.trim(), role);
      }
      return user;
    } on FirebaseAuthException catch (e) {
      throw _handleAuthError(e);
    } catch (e) {
      throw 'حدث خطأ أثناء إنشاء الحساب.';
    }
  }

  Future<void> _createUserRecord(String uid, String name, String nickname, String email, String role) async {
    final normalizedRole = role.trim().toLowerCase();
    final Map<String, dynamic> userData = {
      'uid': uid,
      'name': name,
      'nickname': nickname,
      'email': email,
      'userType': normalizedRole,
      'averageRating': 0.0,
      'ratingCount': 0,
      'mobileNumber': '',
      'profilePictureUrl': '',
      'familyStatus': false,
      'childrenCount': 0,
      'isIdVerified': false,
      'identityCardUrl': '',
      'documents': <String, dynamic>{},
      'createdAt': FieldValue.serverTimestamp(),
      'isDriver': normalizedRole == 'driver',
    };

    if (normalizedRole == 'driver') {
      userData.addAll({
        'drivingLicenseNumber': '',
        'plateNumber': '',
        'carOwnerOfficialName': '',
        'driverNotes': '',
        'vehicleIds': <String>[],
        'isActiveDriver': true,
      });
    }

    if (normalizedRole == 'agency') {
      userData.addAll({
        'agencyName': name,
        'agencyVerified': false,
        'agencyVehicles': <String>[],
      });
    }

    await _firestore.collection('users').doc(uid).set(userData);
  }

  Future<AppUser?> getUserData(String uid) async {
    try {
      final doc = await _firestore.collection('users').doc(uid).get();
      if (doc.exists) {
        return AppUser.fromDocument(doc);
      }
      return null;
    } catch (e) {
      debugPrint('AuthService.getUserData error: $e');
      return null;
    }
  }

  Future<bool> updateUserRecord(String uid, Map<String, dynamic> updates) async {
    if (uid.isEmpty) return false;
    try {
      final payload = Map<String, dynamic>.from(updates);
      payload.removeWhere((k, v) => v == null);
      payload['updatedAt'] = FieldValue.serverTimestamp();
      await _firestore.collection('users').doc(uid).update(payload);
      return true;
    } catch (e) {
      debugPrint('AuthService.updateUserRecord error: $e');
      return false;
    }
  }

  Future<bool> deleteUserAccount(String uid) async {
    try {
      await _firestore.collection('users').doc(uid).delete();
      final current = _auth.currentUser;
      if (current != null && current.uid == uid) {
        await current.delete();
      }
      return true;
    } catch (e) {
      debugPrint('AuthService.deleteUserAccount error: $e');
      return false;
    }
  }

  Future<void> signOut() async {
    await _auth.signOut();
  }

  User? getCurrentUser() => _auth.currentUser;
}

import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:uuid/uuid.dart';
import 'package:flutter/foundation.dart';
import '../models/booking_model.dart';
import '../models/trip.dart';
import '../models/app_user.dart';
import 'notification_service.dart';

class BookingService {
  final FirebaseFirestore _firestore = FirebaseFirestore.instance;
  final Uuid _uuid = const Uuid();

  final String _tripsCollection = 'trips';
  final String _bookingsCollection = 'bookings';
  final String _usersCollection = 'users';

  BookingService();

  // ---------- Throttle state for client-side expire checks ----------
  DateTime? _lastExpireCheck;

  /// يفحص ويُلغي الحجوزات المعلقة التي انتهت صلاحيتها من جهة العميل.
  /// لا يستدعى أكثر من مرة خلال `minIntervalMinutes` على الجهاز الواحد.
  /// يستخدم batch لتقليل عدد عمليات الكتابة.
  Future<void> checkAndExpireBookings({int limit = 20, int minIntervalMinutes = 3}) async {
    final now = DateTime.now();
    if (_lastExpireCheck != null) {
      final diff = now.difference(_lastExpireCheck!);
      if (diff.inMinutes < minIntervalMinutes) {
        debugPrint('checkAndExpireBookings: throttled (last run ${diff.inMinutes} minutes ago)');
        return;
      }
    }
    _lastExpireCheck = now;

    final Timestamp nowTs = Timestamp.fromDate(now);

    try {
      final bookingSnapshot = await _firestore
          .collection(_bookingsCollection)
          .where('bookingStatus', isEqualTo: 'pending')
          .where('reservedUntil', isLessThanOrEqualTo: nowTs)
          .limit(limit)
          .get();

      if (bookingSnapshot.docs.isEmpty) {
        debugPrint('No expired pending bookings found (client check).');
        return;
      }

      debugPrint('Found ${bookingSnapshot.docs.length} expired bookings (client). Processing...');

      final WriteBatch batch = _firestore.batch();
      final Map<String, Map<String, dynamic>> tripsToUpdate = {};

      for (final doc in bookingSnapshot.docs) {
        final bookingData = doc.data();
        final String bookingId = doc.id;
        final String? tripId = bookingData['tripId'] as String?;
        final List<dynamic> seatIndexes = bookingData['seatIndexes'] ?? [];
        final int bookedSeatsCount = (bookingData['bookedSeatsCount'] ?? 0) as int;

        // تحديث حالة الحجز إلى ملغى
        batch.update(doc.reference, {
          'bookingStatus': 'cancelled',
          'canceledAt': FieldValue.serverTimestamp(),
          'updatedAt': FieldValue.serverTimestamp(),
          'updatedBy': 'client_expire',
        });

        if (tripId != null && tripId.isNotEmpty) {
          if (!tripsToUpdate.containsKey(tripId)) {
            tripsToUpdate[tripId] = {
              'seatIndexesList': <List<dynamic>>[],
              'totalSeatsToReturn': 0,
            };
          }
          (tripsToUpdate[tripId]!['seatIndexesList'] as List).add(seatIndexes);
          tripsToUpdate[tripId]!['totalSeatsToReturn'] =
              (tripsToUpdate[tripId]!['totalSeatsToReturn'] as int) + bookedSeatsCount;
        }
      }

      // قراءة كل الرحلات المطلوبة وتحديثها في الـ batch
      for (final tripId in tripsToUpdate.keys) {
        final tripRef = _firestore.collection(_tripsCollection).doc(tripId);
        final tripSnap = await tripRef.get();
        if (!tripSnap.exists) {
          debugPrint('Trip $tripId not found while expiring bookings (client).');
          continue;
        }

        final tripData = tripSnap.data() as Map<String, dynamic>;
        List<dynamic> seatStatus = List<dynamic>.from(tripData['seatStatus'] ?? []);
        final int totalSeats = (tripData['totalSeats'] ?? seatStatus.length) as int;

        if (seatStatus.isEmpty) {
          seatStatus = List.generate(totalSeats, (_) => true);
        }

        final List<List<dynamic>> seatIndexesList =
        List<List<dynamic>>.from(tripsToUpdate[tripId]!['seatIndexesList'] as List);
        int totalToReturn = tripsToUpdate[tripId]!['totalSeatsToReturn'] as int;

        bool usedIndexes = false;
        for (final idxList in seatIndexesList) {
          if (idxList.isNotEmpty) {
            usedIndexes = true;
            for (final idx in idxList) {
              if (idx is int && idx >= 0 && idx < seatStatus.length) {
                seatStatus[idx] = true;
              }
            }
          }
        }

        if (!usedIndexes && totalToReturn > 0) {
          int released = 0;
          for (int i = 0; i < seatStatus.length && released < totalToReturn; i++) {
            if (seatStatus[i] == false) {
              seatStatus[i] = true;
              released++;
            }
          }
        }

        batch.update(tripRef, {
          'seatStatus': seatStatus,
          'availableSeats': FieldValue.increment(totalToReturn),
          'updatedAt': FieldValue.serverTimestamp(),
        });
      }

      await batch.commit();
      debugPrint('Expired bookings processed and trips updated successfully (client).');
    } catch (e, st) {
      debugPrint('Error in checkAndExpireBookings: $e\n$st');
    }
  }

  // ---------- Existing methods (kept and unchanged except minor integration) ----------

  /// إنشاء حجز مؤقت (pending) مع دعم idempotency عبر clientBookingId
  /// يعيد bookingId عند النجاح أو رسالة خطأ (String) عند الفشل
  Future<String?> createBookingPending({
    required String tripId,
    required String passengerId,
    required int seatsToBook,
    required double tripPricePerSeat,
    bool isFamilyBooking = false,
    int childrenCount = 0,
    String? clientBookingId,
    String? notes,
    Duration reservationDuration = const Duration(minutes: 15),
  }) async {
    if (seatsToBook <= 0) return 'عدد المقاعد غير صالح.';

    String? errorMessage;
    String? bookingIdForReturn;

    try {
      // جلب بيانات الراكب (لملء passengerNickname)
      final passengerDoc = await _firestore.collection(_usersCollection).doc(passengerId).get();
      if (!passengerDoc.exists) return 'بيانات الراكب غير موجودة.';
      final appUser = AppUser.fromDocument(passengerDoc);

      // نستخدم clientBookingId إن وُجد لتفادي التكرار
      final bookingId = clientBookingId ?? _uuid.v4();
      final bookingRef = _firestore.collection(_bookingsCollection).doc(bookingId);

      // تحقق مبكر من وجود مستند بنفس المعرف (idempotency)
      final existingBooking = await bookingRef.get();
      if (existingBooking.exists) {
        // إذا وُجد مستند بنفس id نعيده كنجاح idempotent
        return bookingRef.id;
      }

      // تحقق سريع من وجود حجز قائم لنفس الراكب والرحلة (confirmed أو pending)
      final existingQuery = await _firestore
          .collection(_bookingsCollection)
          .where('tripId', isEqualTo: tripId)
          .where('passengerId', isEqualTo: passengerId)
          .where('bookingStatus', whereIn: ['pending', 'confirmed'])
          .limit(1)
          .get();

      if (existingQuery.docs.isNotEmpty) {
        return 'لديك حجز قائم لهذه الرحلة.';
      }

      // ابدأ الترانزاكشن لحجز المقاعد وكتابة الحجز في نفس العملية
      await _firestore.runTransaction((transaction) async {
        final tripRef = _firestore.collection(_tripsCollection).doc(tripId);
        final tripSnapshot = await transaction.get(tripRef);

        if (!tripSnapshot.exists) {
          errorMessage = 'الرحلة غير موجودة.';
          return;
        }

        final trip = Trip.fromDocument(tripSnapshot);

        if (trip.availableSeats < seatsToBook) {
          errorMessage = 'لا توجد مقاعد كافية. المتاح حالياً: ${trip.availableSeats}';
          return;
        }

        // حجز المقاعد: نحجز أول N مقاعد متاحة ونحصل على seatIndexes
        final newSeatStatus = _reserveSeats(trip.seatStatus, seatsToBook);
        if (newSeatStatus == null) {
          errorMessage = 'فشل في تخصيص المقاعد.';
          return;
        }

        final seatIndexes = _extractReservedIndexes(trip.seatStatus, newSeatStatus);
        final totalPrice = seatsToBook * tripPricePerSeat;
        final reservedUntil = Timestamp.fromDate(DateTime.now().add(reservationDuration));

        final bookingPayload = {
          'tripId': tripId,
          'passengerId': passengerId,
          'passengerNickname': appUser.nickname ?? '',
          'driverId': trip.driverId,
          'bookedSeatsCount': seatsToBook,
          'totalPrice': totalPrice,
          'isFamilyBooking': isFamilyBooking,
          'childrenCount': childrenCount,
          'bookingStatus': 'pending',
          'paymentStatus': 'unpaid',
          'notes': notes ?? '',
          'seatIndexes': seatIndexes,
          'reservedUntil': reservedUntil,
          'paymentIntentId': null,
          'canceledAt': null,
          'createdAt': FieldValue.serverTimestamp(),
          'updatedAt': FieldValue.serverTimestamp(),
        };

        transaction.set(bookingRef, bookingPayload);

        transaction.update(tripRef, {
          'seatStatus': newSeatStatus,
          'availableSeats': FieldValue.increment(-seatsToBook),
          'updatedAt': FieldValue.serverTimestamp(),
        });

        bookingIdForReturn = bookingRef.id;
      });

      if (errorMessage == null && bookingIdForReturn != null) {
        _sendSafeNotification(passengerId, 'تم إنشاء الحجز مؤقتاً', 'تم حجز المقاعد مؤقتاً، يرجى إتمام الدفع خلال المهلة المحددة.');
      }

      return errorMessage ?? bookingIdForReturn;
    } catch (e, st) {
      debugPrint('BookingService.createBookingPending error: $e\n$st');
      return 'حدث خطأ أثناء إنشاء الحجز.';
    }
  }

  /// تأكيد الحجز بعد نجاح الدفع: يحدث bookingStatus -> confirmed و paymentStatus -> paid
  /// يتحقق من أن الحجز ما زال pending ولم تنتهِ reservedUntil
  Future<String?> confirmBookingAfterPayment({
    required String bookingId,
    required String paymentIntentId,
    required String payerId,
  }) async {
    try {
      final bookingRef = _firestore.collection(_bookingsCollection).doc(bookingId);
      final bookingSnap = await bookingRef.get();
      if (!bookingSnap.exists) return 'الحجز غير موجود.';

      final booking = Booking.fromDocument(bookingSnap);

      if (booking.passengerId != payerId) return 'غير مصرح لك بتأكيد هذا الحجز.';

      final raw = bookingSnap.data() ?? {};
      final reservedUntilRaw = raw['reservedUntil'];
      Timestamp? reservedUntil;
      if (reservedUntilRaw is Timestamp) reservedUntil = reservedUntilRaw;
      else if (reservedUntilRaw is String) {
        final dt = DateTime.tryParse(reservedUntilRaw);
        if (dt != null) reservedUntil = Timestamp.fromDate(dt);
      }

      if (booking.bookingStatus.toLowerCase() == 'confirmed') return null;

      if (reservedUntil != null && reservedUntil.toDate().isBefore(DateTime.now())) {
        // الحجز انتهت مهلة الحجز المؤقت
        // نلغي الحجز بدلاً من تأكيده
        await cancelBooking(bookingId: bookingId, passengerId: payerId);
        return 'انتهت مهلة الحجز المؤقت، تم إلغاؤه.';
      }

      await bookingRef.update({
        'bookingStatus': 'confirmed',
        'paymentStatus': 'paid',
        'paymentIntentId': paymentIntentId,
        'updatedAt': FieldValue.serverTimestamp(),
      });

      _sendSafeNotification(booking.passengerId, 'تم تأكيد الحجز', '✅ تم تأكيد حجزك بنجاح.');

      return null;
    } catch (e, st) {
      debugPrint('BookingService.confirmBookingAfterPayment error: $e\n$st');
      return 'فشل تأكيد الحجز بعد الدفع.';
    }
  }

  /// إلغاء الحجز (pending أو confirmed) وإرجاع المقاعد
  Future<String?> cancelBooking({
    required String bookingId,
    required String passengerId,
  }) async {
    String? errorMessage;
    try {
      await _firestore.runTransaction((transaction) async {
        final bookingRef = _firestore.collection(_bookingsCollection).doc(bookingId);
        final bookingSnapshot = await transaction.get(bookingRef);

        if (!bookingSnapshot.exists) {
          errorMessage = 'الحجز غير موجود.';
          return;
        }

        final booking = Booking.fromDocument(bookingSnapshot);
        if (booking.passengerId != passengerId) {
          errorMessage = 'غير مصرح لك بإلغاء هذا الحجز.';
          return;
        }

        final tripRef = _firestore.collection(_tripsCollection).doc(booking.tripId);
        final tripSnapshot = await transaction.get(tripRef);

        if (tripSnapshot.exists) {
          final trip = Trip.fromDocument(tripSnapshot);
          final newSeatStatus = _releaseSpecificIndexes(trip.seatStatus, booking.seatIndexes ?? [], booking.bookedSeatsCount);

          transaction.update(tripRef, {
            'seatStatus': newSeatStatus,
            'availableSeats': FieldValue.increment(booking.bookedSeatsCount),
            'updatedAt': FieldValue.serverTimestamp(),
          });
        }

        transaction.update(bookingRef, {
          'bookingStatus': 'cancelled',
          'paymentStatus': booking.paymentStatus == 'paid' ? booking.paymentStatus : 'unpaid',
          'canceledAt': FieldValue.serverTimestamp(),
          'updatedAt': FieldValue.serverTimestamp(),
        });
      });

      _sendSafeNotification(passengerId, 'تم إلغاء الحجز', 'تم إلغاء حجزك وإرجاع المقاعد.');
      return null;
    } catch (e, st) {
      debugPrint('BookingService.cancelBooking error: $e\n$st');
      return 'فشل إلغاء الحجز.';
    }
  }

  /// دالة تُستدعى من Cloud Function أو مجدول لإلغاء الحجوزات المؤقتة المنتهية
  Future<void> expirePendingBooking(String bookingId) async {
    try {
      final bookingRef = _firestore.collection(_bookingsCollection).doc(bookingId);
      final bookingSnap = await bookingRef.get();
      if (!bookingSnap.exists) return;

      final booking = Booking.fromDocument(bookingSnap);
      if (booking.bookingStatus.toLowerCase() != 'pending') return;

      final raw = bookingSnap.data() ?? {};
      final reservedUntilRaw = raw['reservedUntil'];
      Timestamp? reservedUntil;
      if (reservedUntilRaw is Timestamp) reservedUntil = reservedUntilRaw;
      else if (reservedUntilRaw is String) {
        final dt = DateTime.tryParse(reservedUntilRaw);
        if (dt != null) reservedUntil = Timestamp.fromDate(dt);
      }

      if (reservedUntil == null) return;
      if (reservedUntil.toDate().isBefore(DateTime.now())) {
        await cancelBooking(bookingId: bookingId, passengerId: booking.passengerId);
      }
    } catch (e, st) {
      debugPrint('BookingService.expirePendingBooking error: $e\n$st');
    }
  }

  /// جلب حجوزات الراكب
  Future<List<Booking>> getBookingsForPassenger(String passengerId) async {
    try {
      final snapshot = await _firestore
          .collection(_bookingsCollection)
          .where('passengerId', isEqualTo: passengerId)
          .get();

      final list = snapshot.docs.map((doc) => Booking.fromDocument(doc)).toList();
      list.sort((a, b) {
        final aTs = a.createdAt ?? Timestamp.now();
        final bTs = b.createdAt ?? Timestamp.now();
        return bTs.compareTo(aTs);
      });
      return list;
    } catch (e, st) {
      debugPrint('BookingService.getBookingsForPassenger error: $e\n$st');
      return [];
    }
  }

  // --- دوال مساعدة لإدارة المقاعد ---

  List<bool>? _reserveSeats(List<bool> seatStatus, int count) {
    final updated = List<bool>.from(seatStatus);
    int reserved = 0;
    for (int i = 0; i < updated.length && reserved < count; i++) {
      if (updated[i]) {
        updated[i] = false;
        reserved++;
      }
    }
    return reserved == count ? updated : null;
  }

  List<bool> _releaseSpecificIndexes(List<bool> seatStatus, List<int>? seatIndexes, int count) {
    final updated = List<bool>.from(seatStatus);
    if (seatIndexes != null && seatIndexes.isNotEmpty) {
      for (final idx in seatIndexes) {
        if (idx >= 0 && idx < updated.length) updated[idx] = true;
      }
      return updated;
    }

    int released = 0;
    for (int i = 0; i < updated.length && released < count; i++) {
      if (!updated[i]) {
        updated[i] = true;
        released++;
      }
    }
    return updated;
  }

  List<int> _extractReservedIndexes(List<bool> oldStatus, List<bool> newStatus) {
    final List<int> indexes = [];
    for (int i = 0; i < oldStatus.length && i < newStatus.length; i++) {
      if (oldStatus[i] == true && newStatus[i] == false) {
        indexes.add(i);
      }
    }
    return indexes;
  }

  void _sendSafeNotification(String userId, String title, String body) {
    try {
      NotificationService().sendNotification(userId: userId, title: title, body: body, type: 'booking');
    } catch (e) {
      debugPrint('Notification error: $e');
    }
  }
}

// Utility مستقل لحساب تقسيم المدفوعات والعمولات.
// يُستدعى من payment_service.dart أو أي مكان يحتاج حساب العمولة.
// يعيد CommissionResult يحتوي على النسب والمبالغ المفصّلة.

// الاستخدام النموذجي:
// final result = CommissionService.calculateSplit(

//   govFee: 2.0,
//   agencyId: 'agency123', // أو null للسائق الحر
//   appRateIfDriverFree: 0.04, // 4% للسائق الحر
//   appRateIfAgencyDriver: 0.08, // 8% إذا تابع لمكتب
//   agencyRate: 0.08, // نسبة المكتب من المبلغ (إن وُجد)

// ثم استخدم result.appCommissionAmount, result.driverNetAmount، إلخ.

import 'dart:math';

/// نتيجة حساب العمولة وتقسيم المبلغ
class CommissionResult {
  final double totalAmount; // المبلغ الإجمالي المدفوع من الراكب (قبل أي خصومات)
  final double govFee; // رسوم الحكومة الثابتة (إن وُجدت) تُخصم قبل حساب العمولة أم بعد؟ (نستخدمها كحقل مرجعي)
  final String? agencyId; // null => سائق حر
  final double appCommissionRate; // نسبة التطبيق المطبقة (مثلاً 0.04)
  final double appCommissionAmount; // قيمة العمولة للتطبيق
  final double? agencyCommissionRate; // نسبة المكتب إن وُجد (مثلاً 0.08)
  final double agencyCommissionAmount; // قيمة العمولة للمكتب (0 إذا لا يوجد مكتب)
  final double driverNetAmount; // المبلغ الصافي الذي يستلمه السائق بعد خصم العمولة والمصاريف إن وُجدت
  final double platformReceives; // إجمالي ما يستلمه التطبيق (قد يساوي appCommissionAmount)
  final double roundingDelta; // الفرق الناتج عن التقريب (للتدقيق إن لزم)

  CommissionResult({
    required this.totalAmount,
    required this.govFee,
    required this.agencyId,
    required this.appCommissionRate,
    required this.appCommissionAmount,
    required this.agencyCommissionRate,
    required this.agencyCommissionAmount,
    required this.driverNetAmount,
    required this.platformReceives,
    required this.roundingDelta,
  });

  Map<String, dynamic> toMap() {
    return {
      'totalAmount': totalAmount,
      'govFee': govFee,
      'agencyId': agencyId,
      'appCommissionRate': appCommissionRate,
      'appCommissionAmount': appCommissionAmount,
      'agencyCommissionRate': agencyCommissionRate,
      'agencyCommissionAmount': agencyCommissionAmount,
      'driverNetAmount': driverNetAmount,
      'platformReceives': platformReceives,
      'roundingDelta': roundingDelta,
    };
  }

  @override
  String toString() => toMap().toString();
}

/// خدمة حساب العمولة (Utility)
class CommissionService {
  CommissionService._(); // لا يمكن إنشاء مثيل

  /// تقريب آمن إلى منزلتين عشريتين
  static double _round2(double v) => (v * 100).roundToDouble() / 100.0;

  /// التحقق من صحة النسبة (بين 0 و 1)
  static double _normalizeRate(double? rate, {required double fallback}) {
    if (rate == null) return fallback;
    if (rate.isNaN) return fallback;
    if (rate < 0) return 0.0;
    if (rate > 1) return fallback;
    return rate;
  }

  /// حساب تقسيم المبلغ لرحلة مفردة.
  ///
  /// [totalAmount] = المبلغ الإجمالي المدفوع من الراكب (قد يشمل govFee أو لا).
  /// [govFee] = رسوم الحكومة الثابتة المرتبطة بالرحلة (تُعرض في النتيجة، لا نخصمها تلقائياً من الحساب هنا إلا إذا طلبت ذلك).
  /// [agencyId] = معرف المكتب إن وُجد؛ إذا كان null => سائق حر.
  /// [appRateIfDriverFree] = نسبة التطبيق عندما السائق حر (مثلاً 0.04).
  /// [appRateIfAgencyDriver] = نسبة التطبيق عندما السائق تابع لمكتب (مثلاً 0.08).
  /// [agencyRate] = نسبة المكتب من المبلغ (مثلاً 0.08). إذا null أو 0 => لا يوجد مكتب.
  /// [deductGovFeeBeforeCommission] = إذا true نخصم govFee من المبلغ قبل حساب العمولة (بعض السياسات تفعل ذلك).
  ///
  /// تُعيد CommissionResult مع القيم المقربة إلى منزلتين عشريتين.
  static CommissionResult calculateSplit({
    required double totalAmount,
    double govFee = 0.0,
    String? agencyId,
    double appRateIfDriverFree = 0.04,
    double appRateIfAgencyDriver = 0.08,
    double? agencyRate,
    bool deductGovFeeBeforeCommission = false,
  }) {
    // تأكيد القيم الأساسية
    final safeTotal = totalAmount.isFinite && totalAmount > 0 ? totalAmount : 0.0;
    final safeGovFee = govFee.isFinite && govFee > 0 ? govFee : 0.0;

    // تحديد نسبة التطبيق حسب وجود مكتب أم لا
    final bool hasAgency = agencyId != null && (agencyId.trim().isNotEmpty);
    final double appRate = _normalizeRate(
      hasAgency ? appRateIfAgencyDriver : appRateIfDriverFree,
      fallback: hasAgency ? 0.08 : 0.04,
    );

    final double normalizedAgencyRate = _normalizeRate(agencyRate, fallback: 0.0);

    // المبلغ الذي سيُحتسب عليه العمولة
    final double baseForCommission = deductGovFeeBeforeCommission ? max(0.0, safeTotal - safeGovFee) : safeTotal;

    // حساب مبالغ العمولة
    final double rawAppCommission = baseForCommission * appRate;
    final double rawAgencyCommission = hasAgency ? baseForCommission * normalizedAgencyRate : 0.0;

    // المبلغ الصافي للسائق قبل التقريب
    final double rawDriverNet = safeTotal - rawAppCommission - rawAgencyCommission;

    // تقريب القيم إلى منزلتين عشريتين
    final double appCommissionAmount = _round2(rawAppCommission);
    final double agencyCommissionAmount = _round2(rawAgencyCommission);
    final double driverNetAmount = _round2(rawDriverNet);

    // ما تستلمه المنصة (قد تساوي appCommissionAmount؛ إذا أردت إضافة رسوم بوابة دفع هنا يمكن إضافتها)
    final double platformReceives = appCommissionAmount;

    // حساب الفرق الناتج عن التقريب (لأغراض التدقيق)
    final double roundingDelta = _round2((safeTotal) - (appCommissionAmount + agencyCommissionAmount + driverNetAmount));

    return CommissionResult(
      totalAmount: _round2(safeTotal),
      govFee: _round2(safeGovFee),
      agencyId: agencyId,
      appCommissionRate: appRate,
      appCommissionAmount: appCommissionAmount,
      agencyCommissionRate: hasAgency ? normalizedAgencyRate : null,
      agencyCommissionAmount: agencyCommissionAmount,
      driverNetAmount: driverNetAmount,
      platformReceives: platformReceives,
      roundingDelta: roundingDelta,
    );
  }

  /// حساب تقسيم لمبلغ مقسوم على عدد مقاعد (مثلاً حجز عدة مقاعد)
  ///
  /// يعيد CommissionResult مجمّع للمبلغ الكلي.
  static CommissionResult calculateSplitForSeats({
    required double pricePerSeat,
    required int seatsCount,
    double govFeePerSeat = 0.0,
    String? agencyId,
    double appRateIfDriverFree = 0.04,
    double appRateIfAgencyDriver = 0.08,
    double? agencyRate,
    bool deductGovFeeBeforeCommission = false,
  }) {
    final safeSeats = max(0, seatsCount);
    final totalAmount = _round2(pricePerSeat * safeSeats);
    final totalGovFee = _round2(govFeePerSeat * safeSeats);
    return calculateSplit(
      totalAmount: totalAmount,
      govFee: totalGovFee,
      agencyId: agencyId,
      appRateIfDriverFree: appRateIfDriverFree,
      appRateIfAgencyDriver: appRateIfAgencyDriver,
      agencyRate: agencyRate,
      deductGovFeeBeforeCommission: deductGovFeeBeforeCommission,
    );
  }

  /// دالة مساعدة لإنتاج خريطة جاهزة للحفظ في مستند الدفع (payments collection)
  /// تحوي الحقول الأساسية: appCommissionRate, appCommissionAmount, agencyCommissionRate, agencyCommissionAmount, driverNetAmount
  static Map<String, dynamic> buildPaymentSplitMap(CommissionResult result) {
    return {
      'totalAmount': result.totalAmount,
      'govFee': result.govFee,
      'agencyId': result.agencyId,
      'appCommissionRate': result.appCommissionRate,
      'appCommissionAmount': result.appCommissionAmount,
      'agencyCommissionRate': result.agencyCommissionRate,
      'agencyCommissionAmount': result.agencyCommissionAmount,
      'driverNetAmount': result.driverNetAmount,
      'platformReceives': result.platformReceives,
      'roundingDelta': result.roundingDelta,
      'calculatedAt': DateTime.now().toUtc().toIso8601String(),
    };
  }
}

import 'package:flutter/foundation.dart';
import 'package:cloud_firestore/cloud_firestore.dart';

class FirebaseService {
  final FirebaseFirestore _firestore = FirebaseFirestore.instance;

  Stream<List<T>> listenToCollection<T>({
    required String path,
    required T Function(Map<String, dynamic> doc) fromJson,
    List<List<dynamic>>? whereClauses,
    List<List<String>>? orderBy,
    int? limit,
  }) {
    try {
      CollectionReference col = _firestore.collection(path);
      Query query = col;

      if (whereClauses != null) {
        for (final clause in whereClauses) {
          if (clause.length >= 3) {
            final field = clause[0] as String;
            final op = clause[1] as String;
            final value = clause[2];
            switch (op) {
              case '==':
                query = query.where(field, isEqualTo: value);
                break;
              case '<':
                query = query.where(field, isLessThan: value);
                break;
              case '<=':
                query = query.where(field, isLessThanOrEqualTo: value);
                break;
              case '>':
                query = query.where(field, isGreaterThan: value);
                break;
              case '>=':
                query = query.where(field, isGreaterThanOrEqualTo: value);
                break;
              case 'array-contains':
                query = query.where(field, arrayContains: value);
                break;
              case 'in':
                query = query.where(field, whereIn: value);
                break;
            }
          }
        }
      }

      if (orderBy != null) {
        for (final ord in orderBy) {
          if (ord.isNotEmpty) {
            final field = ord[0];
            final dir = ord.length > 1 && ord[1].toLowerCase() == 'desc' ? false : true;
            query = query.orderBy(field, descending: !dir);
          }
        }
      }

      if (limit != null && limit > 0) {
        query = query.limit(limit);
      }

      return query.snapshots().handleError((error) {
        debugPrint('FirebaseService error on $path: $error');
      }).map((snapshot) {
        final List<T> list = [];
        for (final doc in snapshot.docs) {
          try {
            final data = Map<String, dynamic>.from(doc.data() as Map<String, dynamic>);
            data['_id'] = doc.id;
            data['id'] = doc.id; // نضمن وجود الحقلين لزيادة التوافق الاستباقي للموديلات
            final item = fromJson(data);
            list.add(item);
          } catch (e, st) {
            debugPrint('FirebaseService.parse doc error in $path: $e\n$st');
          }
        }
        return list;
      });
    } catch (e, st) {
      debugPrint('FirebaseService.listenToCollection fatal error for $path: $e\n$st');
      return Stream<List<T>>.value(<T>[]);
    }
  }
}

class GovFeeRule {
  final String id;
  final String origin; // can be '*' or exact name
  final String destination; // can be '*' or exact name
  final bool exact; // if true require exact match, else allow case-insensitive contains
  final bool distanceBased;
  final double fixedAmount;
  final List<DistanceTier> distanceTiers;
  final int priority;
  final bool isActive;
  final Timestamp createdAt;
  final Timestamp? updatedAt;

  GovFeeRule({
    required this.id,
    required this.origin,
    required this.destination,
    required this.exact,
    required this.distanceBased,
    required this.fixedAmount,
    required this.distanceTiers,
    required this.priority,
    required this.isActive,
    required this.createdAt,
    this.updatedAt,
  });

  factory GovFeeRule.fromDoc(DocumentSnapshot doc) {
    final data = doc.data() as Map<String, dynamic>? ?? {};
    final tiersRaw = data['distanceTiers'] as List<dynamic>? ?? [];
    final tiers = tiersRaw.map((e) {
      if (e is Map<String, dynamic>) return DistanceTier.fromMap(e);
      return DistanceTier.fromMap(Map<String, dynamic>.from(e as Map));
    }).toList();
    return GovFeeRule(
      id: doc.id,
      origin: (data['origin'] ?? '*') as String,
      destination: (data['destination'] ?? '*') as String,
      exact: (data['exact'] ?? false) as bool,
      distanceBased: (data['distanceBased'] ?? false) as bool,
      fixedAmount: (data['fixedAmount'] ?? 0.0).toDouble(),
      distanceTiers: tiers,
      priority: (data['priority'] ?? 0) as int,
      isActive: (data['isActive'] ?? true) as bool,
      createdAt: data['createdAt'] as Timestamp? ?? Timestamp.now(),
      updatedAt: data['updatedAt'] as Timestamp?,
    );
  }

  Map<String, dynamic> toMap() {
    return {
      'origin': origin,
      'destination': destination,
      'exact': exact,
      'distanceBased': distanceBased,
      'fixedAmount': fixedAmount,
      'distanceTiers': distanceTiers.map((t) => t.toMap()).toList(),
      'priority': priority,
      'isActive': isActive,
      'createdAt': createdAt,
      'updatedAt': updatedAt,
    };
  }
}

class GovFeesService {
  final FirebaseFirestore _firestore;
  final String _collection = 'gov_fees';
  final String _tripsCollection = 'trips';

  GovFeesService({FirebaseFirestore? firestore}) : _firestore = firestore ?? FirebaseFirestore.instance;

  CollectionReference get _col => _firestore.collection(_collection);

  /// إنشاء قاعدة جديدة لرسوم الفرزة
  Future<String> createRule({
    required String origin,
    required String destination,
    bool exact = false,
    bool distanceBased = false,
    double fixedAmount = 0.0,
    List<DistanceTier>? distanceTiers,
    int priority = 0,
    bool isActive = true,
  }) async {
    final now = Timestamp.now();
    final docRef = await _col.add({
      'origin': origin,
      'destination': destination,
      'exact': exact,
      'distanceBased': distanceBased,
      'fixedAmount': fixedAmount,
      'distanceTiers': (distanceTiers ?? []).map((t) => t.toMap()).toList(),
      'priority': priority,
      'isActive': isActive,
      'createdAt': now,
      'updatedAt': now,
    });
    return docRef.id;
  }

  /// تحديث قاعدة موجودة (جزئي)
  Future<void> updateRule(String id, Map<String, dynamic> updates) async {
    updates.removeWhere((k, v) => v == null);
    updates['updatedAt'] = Timestamp.now();
    await _col.doc(id).update(updates);
  }

  /// حذف قاعدة
  Future<void> deleteRule(String id) async {
    await _col.doc(id).delete();
  }

  /// جلب كل القواعد (قابلة للفلترة)
  Future<List<GovFeeRule>> fetchRules({bool onlyActive = true}) async {
    try {
      Query q = _col.orderBy('priority', descending: true);
      if (onlyActive) q = q.where('isActive', isEqualTo: true);
      final snap = await q.get();
      return snap.docs.map((d) => GovFeeRule.fromDoc(d)).toList();
    } catch (e) {
      debugPrint('GovFeesService.fetchRules error: $e');
      return [];
    }
  }

  /// جلب قاعدة حسب المعرف
  Future<GovFeeRule?> getRuleById(String id) async {
    final doc = await _col.doc(id).get();
    if (!doc.exists) return null;
    return GovFeeRule.fromDoc(doc);
  }

  // ============================
  // منطق الحساب
  // ============================

  /// يطابق origin/destination مع قاعدة معينة.
  /// يدعم '*' كوايلدكارد، و exact=true للمطابقة الحرفية، وإلا يستخدم contains (case-insensitive).
  bool _matchRoute(String ruleOrigin, String ruleDest, bool exact, String origin, String destination) {
    final oRule = ruleOrigin.trim();
    final dRule = ruleDest.trim();
    final o = origin.trim();
    final d = destination.trim();

    bool matchPart(String pattern, String value) {
      if (pattern == '*' || pattern.isEmpty) return true;
      if (exact) return pattern.toLowerCase() == value.toLowerCase();
      return value.toLowerCase().contains(pattern.toLowerCase());
    }

    return matchPart(oRule, o) && matchPart(dRule, d);
  }

  /// اختيار أفضل قاعدة مطابقة لمسار معين (origin,destination,distanceKm)
  /// تعطي الأولوية لقاعدة ذات أولوية أعلى، ثم القاعدة الأكثر تحديداً (exact).
  Future<GovFeeRule?> _findBestRule(String origin, String destination, {double? distanceKm}) async {
    final rules = await fetchRules(onlyActive: true);
    GovFeeRule? best;
    for (final r in rules) {
      try {
        if (_matchRoute(r.origin, r.destination, r.exact, origin, destination)) {
          if (best == null) {
            best = r;
            continue;
          }
          // مقارنة الأولوية
          if (r.priority > best.priority) {
            best = r;
            continue;
          }
          // إذا نفس الأولوية، اختر الأكثر تحديداً (exact true)
          if (r.priority == best.priority) {
            if (r.exact && !best.exact) {
              best = r;
              continue;
            }
            // إذا نفس الدقة والأولوية، اختر القاعدة ذات fixedAmount أو أقرب tier إن كانت distanceBased
            // لا تغيير افتراضي
          }
        }
      } catch (e) {
        debugPrint('GovFeesService._findBestRule match error: $e');
      }
    }
    return best;
  }

  /// حساب قيمة govFee بناءً على القاعدة (إن وُجدت) وإرجاع القيمة (double).
  /// إذا لم توجد قاعدة، يعيد 0.0.
  double _computeFeeFromRule(GovFeeRule rule, {double? distanceKm}) {
    if (!rule.isActive) return 0.0;
    if (rule.distanceBased) {
      final d = distanceKm ?? 0.0;
      for (final tier in rule.distanceTiers) {
        if (d >= tier.minKm && d <= tier.maxKm) {
          return tier.amount;
        }
      }
      // إذا لم يطابق أي نطاق، نستخدم fixedAmount كقيمة افتراضية إن وُجدت
      if (rule.fixedAmount > 0) return rule.fixedAmount;
      return 0.0;
    } else {
      return rule.fixedAmount;
    }
  }

  /// واجهة عامة: احسب قيمة govFee لرحلة (باستخدام origin,destination,distanceKm)
  /// - يبحث عن أفضل قاعدة ويطبقها.
  /// - يعيد قيمة مزدوجة (0.0 إذا لا توجد قاعدة).
  Future<double> calculateGovFee({
    required String origin,
    required String destination,
    double? distanceKm,
  }) async {
    try {
      final rule = await _findBestRule(origin, destination, distanceKm: distanceKm);
      if (rule == null) return 0.0;
      final fee = _computeFeeFromRule(rule, distanceKm: distanceKm);
      return fee;
    } catch (e) {
      debugPrint('GovFeesService.calculateGovFee error: $e');
      return 0.0;
    }
  }

  /// احسب وقم بتحديث حقل govFeeAmount في وثيقة الرحلة (داخل معاملة آمنة).
  /// يعيد true عند النجاح.
  Future<bool> computeAndSetGovFeeForTrip(String tripId) async {
    final tripRef = _firestore.collection(_tripsCollection).doc(tripId);
    try {
      await _firestore.runTransaction((tx) async {
        final snap = await tx.get(tripRef);
        if (!snap.exists) throw Exception('Trip not found');
        final trip = Trip.fromDocument(snap);

        final origin = trip.origin ?? '';
        final destination = trip.destination ?? '';
        final distanceKm = trip.distanceKm;

        final fee = await calculateGovFee(origin: origin, destination: destination, distanceKm: distanceKm);

        tx.update(tripRef, {
          'govFeeAmount': fee,
          'manifestStatus': 'pending', // لا تغيير فعلي، مجرد مثال
          'updatedAt': FieldValue.serverTimestamp(),
        });
      });
      return true;
    } catch (e, st) {
      debugPrint('GovFeesService.computeAndSetGovFeeForTrip error: $e\n$st');
      return false;
    }
  }

  /// احسب وقم بتحديث govFeeAmount لعدة رحلات دفعة واحدة (مثلاً عند تغيير قواعد الفرزة)
  Future<int> recomputeGovFeesForAllTrips({int limit = 500}) async {
    int updated = 0;
    try {
      final snap = await _firestore.collection(_tripsCollection).where('status', isEqualTo: 'active').limit(limit).get();
      for (final doc in snap.docs) {
        final trip = Trip.fromDocument(doc);
        final fee = await calculateGovFee(origin: trip.origin ?? '', destination: trip.destination ?? '', distanceKm: trip.distanceKm);
        if ((trip.govFeeAmount ?? 0.0) != fee) {
          await _firestore.collection(_tripsCollection).doc(trip.id).update({
            'govFeeAmount': fee,
            'updatedAt': FieldValue.serverTimestamp(),
          });
          updated++;
        }
      }
    } catch (e) {
      debugPrint('GovFeesService.recomputeGovFeesForAllTrips error: $e');
    }
    return updated;
  }
}

    final stream = _firestore
        .collection(collectionName)
        .where('userId', isEqualTo: userId)
        .orderBy('createdAt', descending: true)
        .limit(limit)
        .snapshots();

    final controller = StreamController<List<AppNotification>>.broadcast();
    StreamSubscription<QuerySnapshot>? sub;

    sub = stream.listen((snap) {
      try {
        final list = snap.docs.map((d) {
          try {
            return AppNotification.fromDocument(d);
          } catch (err) {
            debugPrint('NotificationService.stream parse error: $err');
            return null;
          }
        }).whereType<AppNotification>().toList();

        controller.add(list);
      } catch (e) {
        debugPrint('NotificationService.stream processing error: $e');
      }
    }, onError: (e) {
      debugPrint('NotificationService.stream error: $e');
      controller.addError(e);
    });

    controller.onCancel = () async {
      await sub?.cancel();
      await controller.close();
    };

    return controller.stream;
  }

class PaymentResult {
  final bool success;
  final String? message;
  final String? transactionId;
  final double? amount;
  final Map<String, dynamic>? raw;

  PaymentResult({
    required this.success,
    this.message,
    this.transactionId,
    this.amount,
    this.raw,
  });

  /// بعض الأكواد كانت تتوقع paymentIntentId؛ نعيد transactionId كقيمة افتراضية
  String? get paymentIntentId => transactionId;

  @override
  String toString() => 'PaymentResult(success:$success, transactionId:$transactionId, message:$message)';
}

/// واجهة مزود الدفع القابلة للتبديل
abstract class PaymentProviderAdapter {
  Future<PaymentResult> process({
    required String bookingId,
    required double amount,
    required String paymentMethod,
    required String currency,
  });

  Future<PaymentResult> verify({required String transactionId});
}

/// مزود محاكاة للاختبار المحلي
class MockProviderAdapter implements PaymentProviderAdapter {
  final double failureRate;
  final int minDelayMs;
  final int maxDelayMs;
  final Random _rnd = Random();

  MockProviderAdapter({
    this.failureRate = 0.0,
    this.minDelayMs = 600,
    this.maxDelayMs = 1000,
  }) : assert(failureRate >= 0 && failureRate <= 1);

  Future<void> _maybeDelay() async {
    final ms = minDelayMs + _rnd.nextInt(max(1, maxDelayMs - minDelayMs + 1));
    await Future.delayed(Duration(milliseconds: ms));
  }

  @override
  Future<PaymentResult> process({
    required String bookingId,
    required double amount,
    required String paymentMethod,
    required String currency,
  }) async {
    await _maybeDelay();

    if (amount <= 0) {
      return PaymentResult(success: false, message: 'المبلغ غير صالح', amount: amount);
    }
    if (paymentMethod.toLowerCase().contains('fail')) {
      return PaymentResult(success: false, message: 'محاكاة فشل الدفع من المزود', amount: amount);
    }

    if (_rnd.nextDouble() < failureRate) {
      return PaymentResult(success: false, message: 'محاكاة فشل عشوائي من المزود', amount: amount);
    }

    final txId = 'mock-${const Uuid().v4()}';
    final raw = {
      'provider': 'mock',
      'status': 'success',
      'transactionId': txId,
      'amount': amount,
      'currency': currency,
      'paymentMethod': paymentMethod,
      'bookingId': bookingId,
      'timestamp': DateTime.now().toIso8601String(),
    };

    return PaymentResult(
      success: true,
      message: 'محاكاة نجاح الدفع',
      transactionId: txId,
      amount: amount,
      raw: raw,
    );
  }

  @override
  Future<PaymentResult> verify({required String transactionId}) async {
    await _maybeDelay();
    final success = transactionId.startsWith('mock-') && _rnd.nextDouble() >= failureRate;
    return PaymentResult(
      success: success,
      message: success ? 'محاكاة: تم التحقق من المعاملة' : 'محاكاة: المعاملة غير مؤكدة',
      transactionId: transactionId,
    );
  }
}

/// خدمة الدفع الأساسية
class PaymentService {
  final FirebaseFirestore _firestore;
  final NotificationService _notificationService;
  final BookingService _bookingService;
  final AgenciesService _agenciesService; // اسم موحّد ومهيأ في الـ ctor
  final Uuid _uuid = const Uuid();

  /// اسماء المجموعات
  final String _bookingsCollection = 'bookings';
  final String _paymentsCollection = 'payments';
  final String _usersCollection = 'users';

  /// مزود الدفع القابل للتبديل (يمكن استبداله بمزود حقيقي)
  PaymentProviderAdapter providerAdapter = MockProviderAdapter();

  PaymentService({
    FirebaseFirestore? firestore,
    NotificationService? notificationService,
    BookingService? bookingService,
    AgenciesService? agenciesService,
  })  : _firestore = firestore ?? FirebaseFirestore.instance,
        _notificationService = notificationService ?? NotificationService(),
        _bookingService = bookingService ?? BookingService(),
        _agenciesService = agenciesService ?? AgenciesService();

  // ============================
  // Helpers
  // ============================

  double _toDoubleSafe(dynamic v) {
    if (v == null) return 0.0;
    if (v is double) return v;
    if (v is int) return v.toDouble();
    if (v is num) return v.toDouble();
    if (v is String) return double.tryParse(v) ?? 0.0;
    return 0.0;
  }

  // ============================
  // معالجة الدفع العامّة
  // ============================
  Future<PaymentResult> processPayment({
    required String bookingId,
    required double amount,
    required String paymentMethod, // 'wallet' أو اسم مزود/deeplink
    String currency = 'YER',
    bool useWalletFirst = false,
    double appRateIfDriverFree = 0.04,
    double appRateIfAgencyDriver = 0.08,
    double? agencyRate,
    double govFee = 0.0,
    bool deductGovFeeBeforeCommission = false,
  }) async {
    try {
      final bookingRef = _firestore.collection(_bookingsCollection).doc(bookingId);
      final bookingSnap = await bookingRef.get();

      if (!bookingSnap.exists) {
        return PaymentResult(success: false, message: 'الحجز غير موجود.');
      }

      final booking = Booking.fromDocument(bookingSnap);

      // حاول استخراج agencyId من بيانات المستند مباشرة (لأن نموذج Booking قد لا يحتوي عليه)
      final bookingData = bookingSnap.data() ?? {};
      final agencyIdFromDoc = bookingData['agencyId'] ?? bookingData['agency_id'];
      final String? agencyId = agencyIdFromDoc != null ? agencyIdFromDoc.toString() : null;

      // تحقق مبسط للمبلغ
      if (booking.totalPrice > 0 && (booking.totalPrice - amount).abs() > 0.01) {
        return PaymentResult(success: false, message: 'المبلغ المدفوع لا يطابق مبلغ الحجز.', amount: amount);
      }

      // إذا طلب الدفع من المحفظة فقط
      if (paymentMethod.toLowerCase() == 'wallet') {
        final walletResult = await _processWalletPayment(
          booking: booking,
          amount: amount,
          currency: currency,
          appRateIfDriverFree: appRateIfDriverFree,
          appRateIfAgencyDriver: appRateIfAgencyDriver,
          agencyRate: agencyRate,
          govFee: govFee,
          deductGovFeeBeforeCommission: deductGovFeeBeforeCommission,
          agencyId: agencyId,
        );
        return walletResult;
      }

      // خيار: استخدام المحفظة أولاً ثم الباقي عبر مزود خارجي
      if (useWalletFirst) {
        final userDoc = await _firestore.collection(_usersCollection).doc(booking.passengerId).get();
        final userData = userDoc.data() ?? {};
        final walletBalance = _toDoubleSafe(userData['walletBalance']);
        if (walletBalance >= amount) {
          final walletResult = await _processWalletPayment(
            booking: booking,
            amount: amount,
            currency: currency,
            appRateIfDriverFree: appRateIfDriverFree,
            appRateIfAgencyDriver: appRateIfAgencyDriver,
            agencyRate: agencyRate,
            govFee: govFee,
            deductGovFeeBeforeCommission: deductGovFeeBeforeCommission,
            agencyId: agencyId,
          );
          return walletResult;
        }
      }

      // معالجة الدفع عبر مزود خارجي
      final providerResult = await providerAdapter.process(
        bookingId: bookingId,
        amount: amount,
        paymentMethod: paymentMethod,
        currency: currency,
      );

      if (!providerResult.success) {
        return providerResult;
      }

      final transactionId = providerResult.transactionId ?? _uuid.v4();

      // سجل الدفع في Firestore وتوزيع الأموال داخل معاملة آمنة
      final recordResult = await _recordAndDistributePayment(
        booking: booking,
        amount: amount,
        currency: currency,
        paymentMethod: paymentMethod,
        transactionId: transactionId,
        rawProviderResponse: providerResult.raw,
        appRateIfDriverFree: appRateIfDriverFree,
        appRateIfAgencyDriver: appRateIfAgencyDriver,
        agencyRate: agencyRate,
        govFee: govFee,
        deductGovFeeBeforeCommission: deductGovFeeBeforeCommission,
        agencyId: agencyId,
      );

      if (!recordResult.success) {
        try {
          await _bookingService.cancelBooking(bookingId: booking.id, passengerId: booking.passengerId);
        } catch (e) {
          debugPrint('PaymentService: failed to cancel booking after distribution failure: $e');
        }
        return PaymentResult(success: false, message: recordResult.message);
      }

      // تأكيد الحجز على الخادم عبر BookingService
      final confirmError = await _booking_service_confirmSafe(
        bookingId: booking.id,
        paymentIntentId: transactionId,
        payerId: booking.passengerId,
      );

      if (confirmError != null) {
        return PaymentResult(success: false, message: confirmError);
      }

      // تحديث بيانات الحجز
      await bookingRef.update({
        'paidAt': FieldValue.serverTimestamp(),
        'paymentMethod': paymentMethod,
        'transactionId': transactionId,
        'amountPaid': amount,
        'paymentStatus': 'paid',
        'updatedAt': FieldValue.serverTimestamp(),
      });

      // إشعار الراكب
      try {
        await _notificationService.sendNotification(
          userId: booking.passengerId,
          title: 'تم استلام الدفع',
          body: '✅ تم استلام دفعتك للحجز بمبلغ ${amount.toStringAsFixed(2)} $currency.',
          type: 'payment',
        );
      } catch (e) {
        debugPrint('PaymentService.notification error: $e');
      }

      return PaymentResult(
        success: true,
        message: providerResult.message ?? 'تم الدفع بنجاح',
        transactionId: transactionId,
        amount: amount,
        raw: providerResult.raw,
      );
    } catch (e, st) {
      debugPrint('PaymentService.processPayment error: $e\n$st');
      return PaymentResult(success: false, message: 'حدث خطأ أثناء معالجة الدفع.');
    }
  }

  // ============================
  // دفع من المحفظة (wallet)
  // ============================
  Future<PaymentResult> _processWalletPayment({
    required Booking booking,
    required double amount,
    required String currency,
    double appRateIfDriverFree = 0.04,
    double appRateIfAgencyDriver = 0.08,
    double? agencyRate,
    double govFee = 0.0,
    bool deductGovFeeBeforeCommission = false,
    String? agencyId,
  }) async {
    final passengerRef = _firestore.collection(_usersCollection).doc(booking.passengerId);
    final driverRef = _firestore.collection(_usersCollection).doc(booking.driverId);

    final txId = 'wallet-${_uuid.v4()}';

    try {
      await _firestore.runTransaction((tx) async {
        final passengerSnap = await tx.get(passengerRef);
        if (!passengerSnap.exists) throw Exception('المستخدم غير موجود');

        final passengerData = passengerSnap.data() ?? {};
        final passengerBalance = _toDoubleSafe(passengerData['walletBalance']);

        if (passengerBalance < amount) throw Exception('رصيد المحفظة غير كافٍ');

        final newPassengerBalance = passengerBalance - amount;
        tx.update(passengerRef, {'walletBalance': newPassengerBalance, 'updatedAt': FieldValue.serverTimestamp()});

        final commission = CommissionService.calculateSplit(
          totalAmount: amount,
          govFee: govFee,
          agencyId: agencyId,
          appRateIfDriverFree: appRateIfDriverFree,
          appRateIfAgencyDriver: appRateIfAgencyDriver,
          agencyRate: agencyRate,
          deductGovFeeBeforeCommission: deductGovFeeBeforeCommission,
        );

        final paymentDocRef = _firestore.collection(_paymentsCollection).doc();
        final paymentRecord = {
          'bookingId': booking.id,
          'passengerId': booking.passengerId,
          'driverId': booking.driverId,
          'agencyId': agencyId,
          'amount': commission.totalAmount,
          'currency': currency,
          'paymentMethod': 'wallet',
          'transactionId': txId,
          'status': 'paid',
          'split': CommissionService.buildPaymentSplitMap(commission),
          'rawProvider': {'provider': 'wallet'},
          'createdAt': FieldValue.serverTimestamp(),
          'updatedAt': FieldValue.serverTimestamp(),
        };
        tx.set(paymentDocRef, paymentRecord);

        if (agencyId != null && agencyId.trim().isNotEmpty) {
          final agencyRef = _firestore.collection('agencies').doc(agencyId);
          final agencySnap = await tx.get(agencyRef);
          if (!agencySnap.exists) throw Exception('المكتب غير موجود');
          final agencyData = agencySnap.data() ?? {};
          final currentAgencyBalance = _toDoubleSafe(agencyData['walletBalance']);
          final newAgencyBalance = currentAgencyBalance + commission.agencyCommissionAmount;
          tx.update(agencyRef, {'walletBalance': newAgencyBalance, 'updatedAt': FieldValue.serverTimestamp()});
        } else {
          final driverSnap = await tx.get(driverRef);
          if (!driverSnap.exists) throw Exception('السائق غير موجود');
          final driverData = driverSnap.data() ?? {};
          final currentDriverBalance = _toDoubleSafe(driverData['walletBalance']);
          final newDriverBalance = currentDriverBalance + commission.driverNetAmount;
          tx.update(driverRef, {'walletBalance': newDriverBalance, 'updatedAt': FieldValue.serverTimestamp()});
        }

        tx.update(_firestore.collection(_bookingsCollection).doc(booking.id), {
          'paymentStatus': 'paid',
          'transactionId': txId,
          'amountPaid': amount,
          'updatedAt': FieldValue.serverTimestamp(),
        });
      });

      final confirmError = await _booking_service_confirmSafe(
        bookingId: booking.id,
        paymentIntentId: txId,
        payerId: booking.passengerId,
      );

      if (confirmError != null) {
        return PaymentResult(success: false, message: confirmError);
      }

      try {
        await _notificationService.sendNotification(
          userId: booking.passengerId,
          title: 'تم الدفع من المحفظة',
          body: '✅ تم خصم ${amount.toStringAsFixed(2)} من محفظتك للحجز ${booking.id}.',
          type: 'payment',
        );
      } catch (e) {
        debugPrint('PaymentService.notification error: $e');
      }

      return PaymentResult(success: true, message: 'تم الدفع من المحفظة بنجاح', transactionId: txId, amount: amount);
    } catch (e, st) {
      debugPrint('PaymentService._processWalletPayment error: $e\n$st');
      return PaymentResult(success: false, message: e.toString());
    }
  }

  // ============================
  // تسجيل وتوزيع الدفع (للمزود الخارجي)
  // ============================
  Future<PaymentResult> _recordAndDistributePayment({
    required Booking booking,
    required double amount,
    required String currency,
    required String paymentMethod,
    required String transactionId,
    Map<String, dynamic>? rawProviderResponse,
    double appRateIfDriverFree = 0.04,
    double appRateIfAgencyDriver = 0.08,

    final bookingRef = _firestore.collection(_bookingsCollection).doc(booking.id);
    final driverRef = _firestore.collection(_usersCollection).doc(booking.driverId);
    final agencyRef = agencyId != null ? _firestore.collection('agencies').doc(agencyId) : null;

    try {
      await _firestore.runTransaction((tx) async {
        final bookingSnap = await tx.get(bookingRef);
        if (!bookingSnap.exists) throw Exception('الحجز غير موجود داخل المعاملة');

        final commission = CommissionService.calculateSplit(
          totalAmount: amount,
          govFee: govFee,
          agencyId: agencyId,
          appRateIfDriverFree: appRateIfDriverFree,
          appRateIfAgencyDriver: appRateIfAgencyDriver,
          agencyRate: agencyRate,
          deductGovFeeBeforeCommission: deductGovFeeBeforeCommission,
        );

        final paymentDocRef = _firestore.collection(_paymentsCollection).doc();
        final paymentRecord = {
          'bookingId': booking.id,
          'passengerId': booking.passengerId,
          'driverId': booking.driverId,
          'agencyId': agencyId,
          'amount': commission.totalAmount,
          'currency': currency,
          'paymentMethod': paymentMethod,
          'transactionId': transactionId,
          'status': 'paid',
          'split': CommissionService.buildPaymentSplitMap(commission),
          'rawProvider': rawProviderResponse ?? {},
          'createdAt': FieldValue.serverTimestamp(),
          'updatedAt': FieldValue.serverTimestamp(),
        };
        tx.set(paymentDocRef, paymentRecord);

        if (agencyRef != null) {
          final agencySnap = await tx.get(agencyRef);
          if (!agencySnap.exists) throw Exception('المكتب غير موجود');
          final agencyData = agencySnap.data() ?? {};
          final currentAgencyBalance = _toDoubleSafe(agencyData['walletBalance']);
          final newAgencyBalance = currentAgencyBalance + commission.agencyCommissionAmount;
          tx.update(agencyRef, {'walletBalance': newAgencyBalance, 'updatedAt': FieldValue.serverTimestamp()});
        }

        final driverSnap = await tx.get(driverRef);
        if (!driverSnap.exists) throw Exception('السائق غير موجود');
        final driverData = driverSnap.data() ?? {};
        final currentDriverBalance = _toDoubleSafe(driverData['walletBalance']);
        final newDriverBalance = currentDriverBalance + commission.driverNetAmount;
        tx.update(driverRef, {'walletBalance': newDriverBalance, 'updatedAt': FieldValue.serverTimestamp()});

        tx.update(bookingRef, {
          'paymentStatus': 'paid',
          'transactionId': transactionId,
          'amountPaid': amount,
          'updatedAt': FieldValue.serverTimestamp(),
        });
      });

      return PaymentResult(success: true, message: 'تم تسجيل وتوزيع الدفع بنجاح', transactionId: transactionId, amount: amount);
    } catch (e, st) {
      debugPrint('PaymentService._recordAndDistributePayment error: $e\n$st');
      return PaymentResult(success: false, message: 'فشل تسجيل/توزيع الدفع: $e');
    }
  }

  // ============================
  // استرجاع الأموال (Refund) - مبسّط
  // ============================
  Future<PaymentResult> refundPayment({
    String? transactionId,
    String? bookingId,
    required String reason,
  }) async {
    try {
      QuerySnapshot snap;
      if (transactionId != null) {
        snap = await _firestore.collection(_paymentsCollection).where('transactionId', isEqualTo: transactionId).limit(1).get();
      } else if (bookingId != null) {
        snap = await _firestore.collection(_paymentsCollection).where('bookingId', isEqualTo: bookingId).limit(1).get();
      } else {
        return PaymentResult(success: false, message: 'مطلوب transactionId أو bookingId لبدء الاسترجاع');
      }

      if (snap.docs.isEmpty) return PaymentResult(success: false, message: 'سجل الدفع غير موجود');

      final doc = snap.docs.first;
      final data = doc.data() as Map<String, dynamic>;
      final status = (data['status'] ?? '') as String;
      if (status == 'refunded') return PaymentResult(success: false, message: 'تم استرجاع هذه المعاملة سابقاً');

      final paymentId = doc.id;
      final bookingIdFound = data['bookingId'] as String?;
      final passengerId = data['passengerId'] as String?;
      final driverId = data['driverId'] as String?;
      final agencyId = data['agencyId'] as String?;
      final amount = _toDoubleSafe(data['amount']);
      final paymentMethod = (data['paymentMethod'] ?? 'unknown') as String;

      if (paymentMethod == 'wallet') {
        await _firestore.runTransaction((tx) async {
          final paymentRef = _firestore.collection(_paymentsCollection).doc(paymentId);
          final passengerRef = _firestore.collection(_usersCollection).doc(passengerId);
          final driverRef = _firestore.collection(_usersCollection).doc(driverId);
          final agencyRef = agencyId != null ? _firestore.collection('agencies').doc(agencyId) : null;

          final passengerSnap = await tx.get(passengerRef);
          if (!passengerSnap.exists) throw Exception('المستخدم غير موجود');
          final passengerData = passengerSnap.data() ?? {};
          final passengerBalance = _toDoubleSafe(passengerData['walletBalance']);
          tx.update(passengerRef, {'walletBalance': passengerBalance + amount, 'updatedAt': FieldValue.serverTimestamp()});

          if (agencyRef != null) {
            final agencySnap = await tx.get(agencyRef);
            if (!agencySnap.exists) throw Exception('المكتب غير موجود');
            final agencyData = agencySnap.data() ?? {};
            final agencyBalance = _toDoubleSafe(agencyData['walletBalance']);
            final agencyNew = max<double>(0.0, agencyBalance - amount);
            tx.update(agencyRef, {'walletBalance': agencyNew, 'updatedAt': FieldValue.serverTimestamp()});
          } else {
            final driverSnap = await tx.get(driverRef);
            if (!driverSnap.exists) throw Exception('السائق غير موجود');
            final driverData = driverSnap.data() ?? {};
            final driverBalance = _toDoubleSafe(driverData['walletBalance']);
            final driverNew = max<double>(0.0, driverBalance - amount);
            tx.update(driverRef, {'walletBalance': driverNew, 'updatedAt': FieldValue.serverTimestamp()});
          }

          tx.update(paymentRef, {
            'status': 'refunded',
            'refundedAt': FieldValue.serverTimestamp(),
            'refundReason': reason,
            'updatedAt': FieldValue.serverTimestamp(),
          });

          if (bookingIdFound != null) {
            final bookingRef = _firestore.collection(_bookingsCollection).doc(bookingIdFound);
            tx.update(bookingRef, {
              'paymentStatus': 'refunded',
              'updatedAt': FieldValue.serverTimestamp(),
            });
          }
        });

        if (passengerId != null) {
          try {
            await _notificationService.sendNotification(
              userId: passengerId,
              title: 'تم استرجاع المبلغ',
              body: 'تم استرجاع مبلغ ${amount.toStringAsFixed(2)} إلى محفظتك.',
              type: 'refund',
            );
          } catch (e) {
            debugPrint('PaymentService.refund notification error: $e');
          }
        }

        return PaymentResult(success: true, message: 'تم استرجاع المبلغ بنجاح', transactionId: transactionId, amount: amount);
      }

      return PaymentResult(success: false, message: 'استرجاع المدفوعات عبر المزود الخارجي غير مفعّل في هذه النسخة. نفّذ الاسترجاع عبر لوحة المزود ثم حدّث سجل الدفع.');
    } catch (e, st) {
      debugPrint('PaymentService.refundPayment error: $e\n$st');
      return PaymentResult(success: false, message: 'حدث خطأ أثناء محاولة الاسترجاع: $e');
    }
  }

  // ============================
  // دوال مساعدة عامة مطلوبة من Providers/واجهات أخرى
  // ============================

  Future<String> getPaymentStatus(String bookingId) async {
    try {
      final snap = await _firestore.collection(_paymentsCollection).where('bookingId', isEqualTo: bookingId).limit(1).get();
      if (snap.docs.isEmpty) return 'unknown';
      final data = snap.docs.first.data();
      return (data['status'] ?? 'unknown') as String;
    } catch (e) {
      debugPrint('PaymentService.getPaymentStatus error: $e');
      return 'unknown';
    }
  }

  Map<String, dynamic> buildDeepLinkPayload({
    required String bookingId,
    required double amount,
    required String paymentMethod,
    required String callbackUrl,
  }) {
    return {
      'bookingId': bookingId,
      'amount': amount,
      'paymentMethod': paymentMethod,
      'callbackUrl': callbackUrl,
      'timestamp': DateTime.now().toIso8601String(),
    };
  }

  String buildDeepLinkUri({required String schemeHost, required Map<String, dynamic> payload}) {
    final encoded = Uri.encodeComponent(jsonEncode(payload));
    return '$schemeHost://pay?payload=$encoded';
  }

  Future<PaymentResult> confirmPayment({
    required String transactionId,
    required String bookingId,
    required double amount,
    String? paymentMethod,
    String currency = 'YER',
    String? payerId, // أصبح اختياريًا لتوافق الاستدعاءات الحالية
  }) async {
    try {
      final verifyResult = await providerAdapter.verify(transactionId: transactionId);
      if (!verifyResult.success) {
        return PaymentResult(success: false, message: 'فشل التحقق من المزود', transactionId: transactionId);
      }

      final existing = await _firestore.collection(_paymentsCollection).where('transactionId', isEqualTo: transactionId).limit(1).get();
      if (existing.docs.isEmpty) {
        await _firestore.collection(_paymentsCollection).add({
          'bookingId': bookingId,
          'transactionId': transactionId,
          'amount': amount,
          'currency': currency,
          'paymentMethod': paymentMethod ?? 'unknown',
          'status': 'paid',
          'createdAt': FieldValue.serverTimestamp(),
          'updatedAt': FieldValue.serverTimestamp(),
        });
      } else {
        await _firestore.collection(_paymentsCollection).doc(existing.docs.first.id).update({
          'status': 'paid',
          'updatedAt': FieldValue.serverTimestamp(),
        });
      }

      final confirmError = await _booking_service_confirmSafe(
        bookingId: bookingId,
        paymentIntentId: transactionId,
        payerId: payerId ?? '',
      );

      if (confirmError != null) {
        return PaymentResult(success: false, message: confirmError, transactionId: transactionId);
      }

      return PaymentResult(success: true, message: 'تم تأكيد الدفع', transactionId: transactionId, amount: amount);
    } catch (e, st) {
      debugPrint('PaymentService.confirmPayment error: $e\n$st');
      return PaymentResult(success: false, message: 'خطأ أثناء تأكيد الدفع', transactionId: transactionId);
    }
  }

  Future<String?> _booking_service_confirmSafe({
    required String bookingId,
    required String paymentIntentId,
    required String payerId,
  }) async {
    try {
      return await _bookingService.confirmBookingAfterPayment(
        bookingId: bookingId,
        paymentIntentId: paymentIntentId,
        payerId: payerId,
      );
    } catch (e, st) {
      debugPrint('PaymentService._booking_service_confirmSafe error: $e\n$st');
      return 'فشل تأكيد الحجز على الخادم.';
    }
  }
}

      : _firestore = firestore ?? FirebaseFirestore.instance,
        _storage = storage ?? FirebaseStorage.instance;

  /// توليد بايتس PDF لكشف الركاب لرحلة واحدة (لا يرفع الملف).
  /// يعيد Uint8List أو يرمي استثناء عند الفشل.
  Future<Uint8List> buildManifestPdfBytes(String tripId) async {
    // جلب بيانات الرحلة
    final tripDoc = await _firestore.collection(_tripsCollection).doc(tripId).get();
    if (!tripDoc.exists) throw Exception('Trip not found: $tripId');
    final trip = Trip.fromDocument(tripDoc);

    // جلب الحجوزات المرتبطة بالرحلة (confirmed أو bookingStatus)
    Query bookingsQuery = _firestore.collection(_bookingsCollection).where('tripId', isEqualTo: tripId);

    // بعض المشاريع تستخدم bookingStatus أو status
    // نحاول جلب الحجوزات المؤكدة أولاً
    final snapConfirmed = await bookingsQuery.where('bookingStatus', isEqualTo: 'confirmed').orderBy('createdAt', descending: false).get();
    QuerySnapshot bookingsSnap;
    if (snapConfirmed.docs.isNotEmpty) {
      bookingsSnap = snapConfirmed;
    } else {
      // احتياطي: بعض المستندات قد تستخدم 'status' أو 'paymentStatus'
      final snapAlt = await bookingsQuery.where('status', isEqualTo: 'confirmed').orderBy('createdAt', descending: false).get();
      bookingsSnap = snapAlt;
    }

    final bookings = bookingsSnap.docs.map((d) => d.data() as Map<String, dynamic>).toList();

    // توليد PDF
    final pw.Document pdfDoc = pw.Document();
    final DateFormat dateFmt = DateFormat('yyyy-MM-dd');
    final DateFormat timeFmt = DateFormat('HH:mm');

    final title = (trip.title != null && trip.title!.isNotEmpty) ? trip.title! : 'كشف ركاب الرحلة ${trip.id}';
    final tripDate = trip.date ?? trip.departureTime;
    final dateText = tripDate != null ? dateFmt.format(tripDate) : '-';
    final timeText = tripDate != null ? timeFmt.format(tripDate) : (trip.time != null && trip.time!.isNotEmpty ? trip.time! : '-');

    pdfDoc.addPage(
      pw.MultiPage(
        pageFormat: pdf.PdfPageFormat.a4,
        margin: const pw.EdgeInsets.all(18),
        build: (pw.Context ctx) {
          return [
            pw.Header(
              level: 0,
              child: pw.Text(
                title,
                style: pw.TextStyle(fontSize: 18, fontWeight: pw.FontWeight.bold),
              ),
            ),
            pw.SizedBox(height: 6),
            pw.Row(
              mainAxisAlignment: pw.MainAxisAlignment.spaceBetween,
              children: [
                pw.Column(crossAxisAlignment: pw.CrossAxisAlignment.start, children: [
                  pw.Text('من: ${trip.origin ?? '-'}', style: pw.TextStyle(fontSize: 12)),
                  pw.Text('إلى: ${trip.destination ?? '-'}', style: pw.TextStyle(fontSize: 12)),
                  pw.Text('التاريخ: $dateText', style: pw.TextStyle(fontSize: 12)),
                  pw.Text('الوقت: $timeText', style: pw.TextStyle(fontSize: 12)),
                ]),
                pw.Column(crossAxisAlignment: pw.CrossAxisAlignment.end, children: [
                  pw.Text('السائق: ${trip.driverName ?? '-'}', style: pw.TextStyle(fontSize: 12)),
                  pw.Text('رقم المركبة: ${trip.carPlateNumber ?? '-'}', style: pw.TextStyle(fontSize: 12)),
                  pw.Text('السعر: ${trip.price?.toStringAsFixed(2) ?? '-'}', style: pw.TextStyle(fontSize: 12)),
                  pw.Text('govFee: ${(trip.govFeeAmount ?? 0.0).toStringAsFixed(2)}', style: pw.TextStyle(fontSize: 12)),
                ]),
              ],
            ),
            pw.SizedBox(height: 12),
            pw.Text('قائمة الركاب', style: pw.TextStyle(fontSize: 14, fontWeight: pw.FontWeight.bold)),
            pw.SizedBox(height: 8),

            // جدول الركاب
            pw.Table.fromTextArray(
              headers: ['#', 'اسم الراكب', 'هاتف', 'مقعد', 'مبلغ مدفوع', 'ملاحظات'],
              data: List<List<String>>.generate(bookings.length, (i) {
                final b = bookings[i];
                final passengerName = (b['passengerName'] ?? b['passengerNickname'] ?? b['name'] ?? '')?.toString() ?? '';
                final phone = (b['passengerPhone'] ?? b['phone'] ?? '')?.toString() ?? '';
                // seat: قد يكون seatIndexes أو seatNumber أو seat
                String seat = '-';
                if (b['seatNumber'] != null) seat = b['seatNumber'].toString();
                else if (b['seatIndexes'] is List && (b['seatIndexes'] as List).isNotEmpty) {
                  final idxList = List.from(b['seatIndexes']);
                  seat = idxList.map((e) => e.toString()).join(',');
                }
                final paid = ((b['amountPaid'] ?? b['amount'] ?? 0).toString());
                final notes = (b['notes'] ?? '')?.toString() ?? '';
                return ['${i + 1}', passengerName, phone, seat, paid, notes];
              }),
              headerStyle: pw.TextStyle(fontWeight: pw.FontWeight.bold),
              cellAlignment: pw.Alignment.centerLeft,
              headerDecoration: pw.BoxDecoration(color: pdf.PdfColors.grey300),
              cellHeight: 20,
              columnWidths: {
                0: pw.FlexColumnWidth(0.6),
                1: pw.FlexColumnWidth(2),
                2: pw.FlexColumnWidth(1.4),
                3: pw.FlexColumnWidth(0.8),
                4: pw.FlexColumnWidth(1.2),
                5: pw.FlexColumnWidth(2),
              },
            ),

            pw.SizedBox(height: 12),
            pw.Row(mainAxisAlignment: pw.MainAxisAlignment.end, children: [
              pw.Column(crossAxisAlignment: pw.CrossAxisAlignment.end, children: [
                pw.Text('إجمالي الركاب: ${bookings.length}', style: pw.TextStyle(fontSize: 12)),
                pw.Text('المقاعد المتاحة: ${trip.availableSeats ?? '-'}', style: pw.TextStyle(fontSize: 12)),
              ])
            ]),

            pw.SizedBox(height: 18),
            pw.Text('ملاحظات إضافية:', style: pw.TextStyle(fontSize: 12, fontWeight: pw.FontWeight.bold)),
            pw.Text(trip.description != null && trip.description!.isNotEmpty ? trip.description! : '-', style: pw.TextStyle(fontSize: 11)),
          ];
        },
      ),
    );

    final Uint8List pdfBytes = await pdfDoc.save();
    return pdfBytes;
  }

  /// توليد PDF ورفع الملف إلى Firebase Storage وتحديث وثيقة الرحلة.
  /// يعيد رابط التحميل (download URL) عند النجاح، أو null عند الفشل.
  Future<String?> generateManifestForTrip(String tripId, {bool attachToTrip = true}) async {
    File? localFile;
    try {
      final Uint8List pdfBytes = await buildManifestPdfBytes(tripId);

      // حفظ PDF مؤقتاً (اختياري، مفيد للتصحيح على الأجهزة المحلية)
      final Directory tempDir = await getTemporaryDirectory();
      final String timestamp = DateFormat('yyyyMMdd_HHmmss').format(DateTime.now());
      final String localPath = '${tempDir.path}/manifest_${tripId}_$timestamp.pdf';
      localFile = File(localPath);
      await localFile.writeAsBytes(pdfBytes, flush: true);

      // رفع الملف إلى Firebase Storage
      final storagePath = '$_manifestsFolder/$tripId/manifest_$timestamp.pdf';
      final ref = _storage.ref().child(storagePath);
      final uploadTask = ref.putData(pdfBytes, SettableMetadata(contentType: 'application/pdf'));

      final snapshot = await uploadTask.whenComplete(() {});
      final downloadUrl = await snapshot.ref.getDownloadURL();

      // تحديث وثيقة الرحلة بحقل manifestUrl ومسار التخزين ووقت الإنشاء
      if (attachToTrip) {
        await _firestore.collection(_tripsCollection).doc(tripId).update({
          'manifestUrl': downloadUrl,
          'manifestStoragePath': storagePath,
          'manifestGeneratedAt': FieldValue.serverTimestamp(),
          'updatedAt': FieldValue.serverTimestamp(),
        });
      }

      // حذف الملف المحلي المؤقت إن وُجد
      try {
        if (localFile.existsSync()) await localFile.delete();
      } catch (_) {}

      return downloadUrl;
    } catch (e, st) {
      debugPrint('PdfService.generateManifestForTrip error: $e\n$st');
      // حاول حذف الملف المحلي عند الفشل
      try {
        if (localFile != null && await localFile.exists()) await localFile.delete();
      } catch (_) {}
      return null;
    }
  }

  /// توليد ورفع manifests لعدة رحلات (متسلسل). يعيد خريطة tripId -> manifestUrl (null عند الفشل).
  Future<Map<String, String?>> generateAndAttachManifestForTrips(List<String> tripIds) async {
    final Map<String, String?> results = {};
    for (final id in tripIds) {
      final url = await generateManifestForTrip(id, attachToTrip: true);
      results[id] = url;
    }
    return results;
  }

  /// حذف ملف الـ manifest من التخزين وتحديث وثيقة الرحلة لإزالة الحقول
  Future<bool> deleteManifest(String tripId) async {
    try {
      final docRef = _firestore.collection(_tripsCollection).doc(tripId);
      final doc = await docRef.get();
      if (!doc.exists) return false;
      final data = doc.data() as Map<String, dynamic>? ?? {};
      final storagePath = (data['manifestStoragePath'] as String?) ?? (data['manifest_path'] as String?);

      if (storagePath != null && storagePath.isNotEmpty) {
        final ref = _storage.ref().child(storagePath);
        try {
          await ref.delete();
        } catch (e) {
          // قد يكون الملف غير موجود؛ لا نوقف العملية بسبب ذلك
          debugPrint('PdfService.deleteManifest: storage delete error: $e');
        }
      }

      await docRef.update({
        'manifestUrl': FieldValue.delete(),
        'manifestStoragePath': FieldValue.delete(),
        'manifestGeneratedAt': FieldValue.delete(),
        'updatedAt': FieldValue.serverTimestamp(),
      });
      return true;
    } catch (e, st) {
      debugPrint('PdfService.deleteManifest error: $e\n$st');
      return false;
    }
  }
}

// lib/services/rating_service.dart
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:flutter/foundation.dart';
import 'package:uuid/uuid.dart';
import '../models/rating_model.dart';

class RatingService {
  final FirebaseFirestore _firestore = FirebaseFirestore.instance;
  final _uuid = const Uuid();
  final String _collection = 'ratings';

  /// إضافة أو تحديث تقييم مع تحديث إحصائيات المستخدم (المُقيَّم) تلقائياً
  /// score يجب أن يكون قيمة رقمية (مثلاً 4.5)
  /// ratingLabel يمكن أن يكون نصًا وصفيًا مثل "excellent" أو "جيد"
  Future<bool> addOrUpdateRating({
    required String tripId,
    required String bookingId,
    required String fromUserId,
    required String toUserId,
    required double score,
    String? ratingLabel,
    String? comment,
    String? note,
  }) async {
    try {
      // 1. التحقق من وجود تقييم سابق من هذا المقيّم لنفس الرحلة/الحجز
      final existing = await _firestore
          .collection(_collection)
          .where('tripId', isEqualTo: tripId)
          .where('bookingId', isEqualTo: bookingId)
          .where('fromUserId', isEqualTo: fromUserId)
          .limit(1)
          .get();

      if (existing.docs.isNotEmpty) {
        // تحديث التقييم الموجود
        final docRef = existing.docs.first.reference;
        final updateData = <String, dynamic>{
          'score': score,
          'rating': ratingLabel ?? (score.toString()),
          'comment': comment,
          'note': note,
          'updatedAt': FieldValue.serverTimestamp(),
        }..removeWhere((k, v) => v == null);

        await docRef.update(updateData);
      } else {
        // إضافة تقييم جديد باستخدام الموديل المحدث
        final ratingId = _uuid.v4();
        final rating = Rating(
          id: ratingId,
          tripId: tripId,
          bookingId: bookingId,
          fromUserId: fromUserId,
          toUserId: toUserId,
          rating: ratingLabel ?? '',
          score: score,
          note: note,
          comment: comment,
          createdAt: Timestamp.now(),
        );

        await _firestore.collection(_collection).doc(ratingId).set(rating.toMap());
      }

      // 2. تحديث إحصائيات المستخدم (المُقيَّم) في مجموعة users
      await _updateUserRatingStats(toUserId);

      return true;
    } catch (e, st) {
      debugPrint('RatingService.addOrUpdateRating error: $e\n$st');
      return false;
    }
  }

  /// دالة داخلية لحساب وتحديث متوسط التقييم وعدد التقييمات في مستند المستخدم
  Future<void> _updateUserRatingStats(String userId) async {
    if (userId.isEmpty) return;
    try {
      // جلب جميع تقييمات هذا المستخدم (التي وجهت إليه)
      final snapshot = await _firestore
          .collection(_collection)
          .where('toUserId', isEqualTo: userId)
          .get();

      if (snapshot.docs.isEmpty) {
        // إذا لم توجد تقييمات، نصفر الإحصائيات
        await _firestore.collection('users').doc(userId).update({
          'averageRating': 0.0,
          'ratingCount': 0,
        });
        return;
      }

      int count = snapshot.docs.length;
      double totalScore = 0.0;

      for (var doc in snapshot.docs) {
        try {
          final ratingObj = Rating.fromDocument(doc);
          totalScore += ratingObj.score;
        } catch (e) {
          debugPrint('RatingService._updateUserRatingStats parse error: $e');
        }
      }

      final double average = count > 0 ? totalScore / count : 0.0;

      // تحديث مستند المستخدم بالحقول الموحدة
      await _firestore.collection('users').doc(userId).update({
        'averageRating': double.parse(average.toStringAsFixed(1)), // تقريب لمنزلة واحدة
        'ratingCount': count,
      });

      debugPrint('Stats updated for $userId: Avg ${average.toStringAsFixed(2)}, Count $count');
    } catch (e, st) {
      debugPrint('RatingService._updateUserRatingStats error: $e\n$st');
    }
  }

  /// جلب التقييمات الموجهة للمستخدم (Stream) لعرضها في الملف الشخصي
  Stream<List<Rating>> getRatingsForUser(String userId) {
    return _firestore
        .collection(_collection)
        .where('toUserId', isEqualTo: userId)
        .snapshots()
        .map((snapshot) {
      final List<Rating> list = [];
      for (final doc in snapshot.docs) {
        try {
          list.add(Rating.fromDocument(doc));
        } catch (e) {
          debugPrint('RatingService.getRatingsForUser parse error: $e');
        }
      }

      // الترتيب حسب التاريخ الأحدث (من الأحدث إلى الأقدم)
      list.sort((a, b) {
        final DateTime? dateA = a.createdAt?.toDate();
        final DateTime? dateB = b.createdAt?.toDate();
        if (dateA == null && dateB == null) return 0;
        if (dateA == null) return 1;
        if (dateB == null) return -1;
        return dateB.compareTo(dateA);
      });

      return list;
    });
  }

  /// جلب متوسط التقييم الحالي من مستند المستخدم مباشرة (للسرعة)
  Future<double> getAverageRatingFromUserDoc(String userId) async {
    try {
      final doc = await _firestore.collection('users').doc(userId).get();
      if (doc.exists) {
        final data = doc.data();
        final val = data?['averageRating'] ?? 0.0;
        if (val is num) return val.toDouble();
        if (val is String) return double.tryParse(val) ?? 0.0;
      }
      return 0.0;
    } catch (e) {
      debugPrint('RatingService.getAverageRatingFromUserDoc error: $e');
      return 0.0;
    }
  }
}

// lib/services/trip_service.dart
//
// خدمة إدارة الرحلات (TripService) مع تكامل لحساب رسوم الفرزة الحكومية (govFee).
// - عند إنشاء رحلة جديدة: تحفظ الرحلة ثم تحسب govFee تلقائياً عبر GovFeesService.
// - عند تحديث رحلة: إذا تغيّر origin أو destination أو distanceKm، تعيد حساب govFee.
// - توفر دوال مساعدة لجلب الرحلات، تحديث المقاعد، حذف الرحلات، وإشعارات المستخدمين.

import 'package:flutter/foundation.dart';
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:uuid/uuid.dart';

import '../models/trip.dart';
import '../models/app_user.dart';
import '../services/notification_service.dart';
import '../services/govfees_service.dart';

class TripService {
  final FirebaseFirestore _firestore;
  final Uuid _uuid = const Uuid();

  final String _tripsCollection = 'trips';
  final String _usersCollection = 'users';
  final String _ratingsCollection = 'ratings';

  final GovFeesService _govFeesService;
  final NotificationService _notificationService; // حقن NotificationService

  TripService({
    FirebaseFirestore? firestore,
    GovFeesService? govFeesService,
    NotificationService? notificationService, // اختياري للـ DI
  })  : _firestore = firestore ?? FirebaseFirestore.instance,
        _govFeesService = govFeesService ?? GovFeesService(),
        _notificationService = notificationService ?? NotificationService();

  /// التحقق من أن المستخدم سائق وموثق (Verified)
  Future<bool> isDriverVerified(String driverId) async {
    try {
      if (driverId.isEmpty) return false;
      final doc = await _firestore.collection(_usersCollection).doc(driverId).get();
      if (!doc.exists) return false;
      final user = AppUser.fromDocument(doc);
      return user.isDriver && user.isIdVerified;
    } catch (e, st) {
      debugPrint('TripService.isDriverVerified error: $e\n$st');
      return false;
    }
  }

  /// إنشاء رحلة جديدة
  /// - يتحقق من توثيق السائق.
  /// - يملأ الحقول الافتراضية (availableSeats, seatStatus, driverName).
  /// - يحفظ الرحلة ثم يستدعي GovFeesService.computeAndSetGovFeeForTrip لحساب govFeeAmount.
  Future<String?> createTrip(Trip trip) async {
    try {
      if (trip.driverId.isEmpty) {
        debugPrint('TripService.createTrip error: driverId is empty');
        return null;
      }

      final isDriverOk = await isDriverVerified(trip.driverId);
      if (!isDriverOk) {
        debugPrint('TripService: Driver is not verified or not a driver.');
        return null;
      }

      // جلب اسم السائق من مستند المستخدم إن لم يكن موجوداً في الكائن
      String driverName = trip.driverName;
      try {
        final userDoc = await _firestore.collection(_usersCollection).doc(trip.driverId).get();
        if (userDoc.exists) {
          final user = AppUser.fromDocument(userDoc);
          if (user.nickname.isNotEmpty) driverName = user.nickname;
        }
      } catch (_) {
        // تجاهل خطأ جلب اسم السائق، نستخدم الاسم الموجود في الكائن
      }

      final String finalId = trip.id.isEmpty ? _uuid.v4() : trip.id;
      final int seats = trip.totalSeats > 0 ? trip.totalSeats : 0;
      final List<bool> initialSeatStatus = List<bool>.filled(seats, true);

      final finalTrip = trip.copyWith(
        id: finalId,
        availableSeats: seats,
        seatStatus: initialSeatStatus,
        driverName: driverName,
        createdAt: Timestamp.now(),
        updatedAt: Timestamp.now(),
      );

      await _firestore.collection(_tripsCollection).doc(finalId).set(finalTrip.toMap());

      // حساب رسوم الفرزة الحكومية تلقائياً بعد الإنشاء
      try {
        await _govFeesService.computeAndSetGovFeeForTrip(finalId);
      } catch (e) {
        debugPrint('TripService.createTrip: gov fee compute failed for $finalId: $e');
        // لا نعيد الخطأ لأن الرحلة تم إنشاؤها بنجاح؛ يمكن إعادة المحاولة لاحقاً
      }

      return finalId;
    } catch (e, st) {
      debugPrint('TripService.createTrip error: $e\n$st');
      return null;
    }
  }

  /// تحديث بيانات الرحلة
  /// - يقارن الحقول القديمة بالجديدة (origin, destination, distanceKm) ويعيد حساب govFee إذا لزم.
  Future<bool> updateTrip(String tripId, Trip updatedTrip) async {
    try {
      final tripRef = _firestore.collection(_tripsCollection).doc(tripId);
      final existingSnap = await tripRef.get();
      if (!existingSnap.exists) {
        debugPrint('TripService.updateTrip: trip not found $tripId');
        return false;
      }

      final existingTrip = Trip.fromDocument(existingSnap);

      // نحضر الخريطة للتحديث
      final data = updatedTrip.toMap();
      // لا نسمح بتغيير createdAt عبر التحديث العادي
      data.remove('createdAt');

      await tripRef.update(data);

      // إذا تغيّر origin أو destination أو distanceKm، أعد حساب govFee
      final originChanged = (existingTrip.origin != updatedTrip.origin);
      final destinationChanged = (existingTrip.destination != updatedTrip.destination);
      final distanceChanged = ((existingTrip.distanceKm ?? 0.0) != (updatedTrip.distanceKm ?? 0.0));

      if (originChanged || destinationChanged || distanceChanged) {
        try {
          await _govFeesService.computeAndSetGovFeeForTrip(tripId);
        } catch (e) {
          debugPrint('TripService.updateTrip: gov fee recompute failed for $tripId: $e');
        }
      }

      return true;
    } catch (e, st) {
      debugPrint('TripService.updateTrip error: $e\n$st');
      return false;
    }
  }

  /// حذف الرحلة
  Future<bool> deleteTrip(String tripId) async {
    try {
      await _firestore.collection(_tripsCollection).doc(tripId).delete();
      return true;
    } catch (e, st) {
      debugPrint('TripService.deleteTrip error: $e\n$st');
      return false;
    }
  }

  /// جلب بيانات رحلة واحدة بواسطة المعرف
  Future<Trip?> getTripById(String tripId) async {
    try {
      final doc = await _firestore.collection(_tripsCollection).doc(tripId).get();
      return doc.exists ? Trip.fromDocument(doc) : null;
    } catch (e, st) {
      debugPrint('TripService.getTripById error: $e\n$st');
      return null;
    }
  }

  /// جلب الرحلات المتاحة للركاب (نشطة + مقاعد شاغرة)
  Future<List<Trip>> getAvailableTrips() async {
    try {
      final snapshot = await _firestore
          .collection(_tripsCollection)
          .where('status', isEqualTo: 'active')
          .where('availableSeats', isGreaterThan: 0)
          .get();

      final list = snapshot.docs.map((doc) => Trip.fromDocument(doc)).toList();

      list.sort((a, b) {
        if (a.departureTime == null) return 1;
        if (b.departureTime == null) return -1;
        return a.departureTime!.compareTo(b.departureTime!);
      });

      return list;
    } catch (e, st) {
      debugPrint('TripService.getAvailableTrips error: $e\n$st');
      return [];
    }
  }

  /// جلب رحلات سائق معين
  Future<List<Trip>> getDriverTrips(String driverId) async {
    try {
      final snapshot = await _firestore
          .collection(_tripsCollection)
          .where('driverId', isEqualTo: driverId)
          .get();

      final list = snapshot.docs.map((doc) => Trip.fromDocument(doc)).toList();

      // ترتيب من الأحدث إلى الأقدم بناءً على تاريخ الإنشاء
      list.sort((a, b) {
        final dateA = a.createdAt?.toDate() ?? DateTime.now();
        final dateB = b.createdAt?.toDate() ?? DateTime.now();
        return dateB.compareTo(dateA);
      });

      debugPrint("TripService: Fetched ${list.length} trips for driver $driverId");
      return list;
    } catch (e, st) {
      debugPrint("TripService.getDriverTrips error: $e\n$st");
      return [];
    }
  }

  /// تحديث حالة المقاعد (يُستدعى عند الحجز)
  Future<bool> updateSeatStatus(String tripId, List<bool> seatStatus, int availableSeats) async {
    try {

    } catch (e, st) {

    }
  }

  /// تغيير حالة الرحلة (active, cancelled, completed)
  Future<void> changeTripStatus(String tripId, String newStatus) async {
    try {

        'updatedAt': FieldValue.serverTimestamp(),
      });
    } catch (e, st) {

    }
  }

  /// إرسال إشعار للمستخدم بخصوص الرحلة
  Future<void> notifyUserAboutTrip(String userId, String title, String body, String type) async {
    try {
      await _notificationService.sendNotification(

        body: body,
        type: type,

    } catch (e, st) {
      debugPrint('TripService.notifyUserAboutTrip error: $e\n$st');
    }
  }

  // ============================

  // ============================

  /// احسب قيمة govFee لرحلة محددة وأعدها (لا يحدث تحديث في Firestore)
  Future<double> calculateGovFeeForTrip(String tripId) async {
    try {
      final trip = await getTripById(tripId);
      if (trip == null) return 0.0;
      final origin = trip.origin;
      final destination = trip.destination;
      final distanceKm = trip.distanceKm;
      final fee = await _govFeesService.calculateGovFee(origin: origin, destination: destination, distanceKm: distanceKm);
      return fee;
    } catch (e) {
      debugPrint('TripService.calculateGovFeeForTrip error: $e');
      return 0.0;
    }
  }

  /// إعادة حساب وتعيين govFeeAmount لرحلة (يدعو GovFeesService.computeAndSetGovFeeForTrip)
  Future<bool> recomputeAndSetGovFeeForTrip(String tripId) async {

class UserService {
  final FirebaseFirestore _firestore;
  final FirebaseStorage? _providedStorage; // قد يكون null في الاختبارات
  final String _collection = 'users';

  // دعم حقن التبعيات للاختبارات والمحاكيات
  // ملاحظة: لا نستدعي FirebaseStorage.instance هنا مباشرةً
  UserService({FirebaseFirestore? firestore, FirebaseStorage? storage})
      : _firestore = firestore ?? FirebaseFirestore.instance,
        _providedStorage = storage;

  // Getter كسلوك lazy للحصول على storage عند الحاجة فقط
  FirebaseStorage get _storage {
    return _providedStorage ?? FirebaseStorage.instance;
  }

  /// جلب بيانات المستخدم وتحويلها إلى كائن AppUser
  Future<AppUser?> getUserById(String uid) async {
    try {
      if (uid.isEmpty) return null;
      final doc = await _firestore.collection(_collection).doc(uid).get();
      return doc.exists ? AppUser.fromDocument(doc) : null;
    } catch (e, st) {
      debugPrint('UserService.getUserById error: $e\n$st');
      return null;
    }
  }

  /// إنشاء مستخدم جديد أو استبدال كامل للمستند (يستخدم toMap من AppUser)
  Future<bool> createOrSetUser(AppUser user) async {
    try {
      if (user.id.isEmpty) {
        debugPrint('UserService.createOrSetUser error: user.id is empty');
        return false;
      }
      final data = user.toMap();
      if (data['createdAt'] == null) data['createdAt'] = FieldValue.serverTimestamp();
      await _firestore.collection(_collection).doc(user.id).set(data, SetOptions(merge: true));
      return true;
    } catch (e, st) {
      debugPrint('UserService.createOrSetUser error: $e\n$st');
      return false;
    }
  }

  /// تحديث الملف الشخصي بالكامل بناءً على الموديل (يحذف الحقول المحمية)
  Future<bool> updateUserProfile(AppUser user) async {
    try {
      if (user.id.isEmpty) {
        debugPrint('UserService.updateUserProfile error: user.id is empty');
        return false;
      }

      final data = user.toMap();
      data.remove('id');
      data.remove('uid');
      data.remove('createdAt');
      data.removeWhere((k, v) => v == null);

      await _firestore.collection(_collection).doc(user.id).update(data);
      return true;
    } catch (e, st) {
      debugPrint('UserService.updateUserProfile error: $e\n$st');
      return false;
    }
  }

  /// تحديث حقول معينة فقط
  Future<bool> updateUserFields(String uid, Map<String, dynamic> fields) async {
    if (uid.isEmpty) return false;
    try {
      fields.removeWhere((k, v) => v == null);
      if (fields.isEmpty) return true;

      if (fields.containsKey('childrenCount')) {
        final v = fields['childrenCount'];
        if (v is String) {
          fields['childrenCount'] = int.tryParse(v) ?? 0;
        } else if (v is double) {
          fields['childrenCount'] = v.toInt();
        }
      }

      await _firestore.collection(_collection).doc(uid).update(fields);
      return true;
    } catch (e, st) {
      debugPrint('UserService.updateUserFields error: $e\n$st');
      return false;
    }
  }

  /// رفع ملف إلى Firebase Storage
  Future<String?> uploadDocument(File file, String uid, String type, {String? extension}) async {
    try {
      final ext = extension ?? file.path.split('.').last;
      final ref = _storage.ref().child('documents/$uid/$type.$ext');

      final UploadTask uploadTask = ref.putFile(file);
      final TaskSnapshot snapshot = await uploadTask.whenComplete(() {});
      return await snapshot.ref.getDownloadURL();
    } catch (e, st) {
      debugPrint('UserService.uploadDocument error: $e\n$st');
      return null;
    }
  }

  /// حفظ رابط الوثيقة في حقل Firestore المناسب
  Future<bool> saveDocumentUrl(String uid, String type, String url) async {
    try {
      if (uid.isEmpty) throw Exception('UID فارغ');

      final userRef = _firestore.collection(_collection).doc(uid);

      if (type == 'profile') {
        await userRef.update({'profilePictureUrl': url});
        return true;
      }

      await _firestore.runTransaction((tx) async {
        final snap = await tx.get(userRef);
        if (!snap.exists) {
          tx.set(userRef, {
            'documents': {type: url},
            'createdAt': FieldValue.serverTimestamp(),
          }, SetOptions(merge: true));
          return;
        }

        final data = snap.data() ?? {};
        final Map<String, dynamic> docs = data['documents'] != null
            ? Map<String, dynamic>.from(data['documents'])
            : <String, dynamic>{};

        docs[type] = url;
        tx.update(userRef, {'documents': docs});
      });

      return true;
    } catch (e, st) {
      debugPrint('UserService.saveDocumentUrl error: $e\n$st');
      return false;
    }
  }

  Future<bool> uploadAndSaveDocument(File file, String uid, String type, {String? extension}) async {
    final url = await uploadDocument(file, uid, type, extension: extension);
    if (url != null) {
      return await saveDocumentUrl(uid, type, url);
    }
    return false;
  }

  Future<bool> deleteDocument(String uid, String type, {String? extension}) async {
    try {
      final ext = extension ?? 'jpg';
      final ref = _storage.ref().child('documents/$uid/$type.$ext');
      await ref.delete();

      final userRef = _firestore.collection(_collection).doc(uid);
      if (type == 'profile') {
        await userRef.update({'profilePictureUrl': FieldValue.delete()});
      } else {
        await _firestore.runTransaction((tx) async {
          final snap = await tx.get(userRef);
          if (!snap.exists) return;
          final data = snap.data() ?? {};
          final Map<String, dynamic> docs = data['documents'] != null
              ? Map<String, dynamic>.from(data['documents'])
              : <String, dynamic>{};
          if (docs.containsKey(type)) {
            docs.remove(type);
            tx.update(userRef, {'documents': docs});
          }
        });
      }

      return true;
    } catch (e, st) {
      debugPrint('UserService.deleteDocument error: $e\n$st');
      return false;
    }
  }

  Future<bool> documentExists(String uid, String type, {String? extension}) async {
    try {
      final ext = extension ?? 'jpg';
      final ref = _storage.ref().child('documents/$uid/$type.$ext');
      await ref.getMetadata();
      return true;
    } catch (_) {
      return false;
    }
  }
}

class VehicleService {
  final CollectionReference _vehiclesRef = FirebaseFirestore.instance.collection('vehicles');

  VehicleService();

  /// جلب مركبات السائق/المالك بناءً على الحقول الجديدة:
  /// يبحث أولاً في حقل vehicleOwner ثم في حقل ownerId كاحتياط
  Future<List<Vehicle>> getDriverVehicles(String driverId) async {
    try {
      // محاولة البحث باستخدام الحقل الموحد vehicleOwner
      final q1 = await _vehiclesRef.where('vehicleOwner', isEqualTo: driverId).get();
      if (q1.docs.isNotEmpty) {
        return q1.docs.map((d) => Vehicle.fromDocument(d)).toList();
      }

      // احتياطي: بعض المستندات القديمة قد تستخدم ownerId أو owner
      final q2 = await _vehiclesRef.where('ownerId', isEqualTo: driverId).get();
      if (q2.docs.isNotEmpty) {
        return q2.docs.map((d) => Vehicle.fromDocument(d)).toList();
      }

      // احتياطي آخر: قد يكون هناك حقل driverId مرتبط بالمركبة
      final q3 = await _vehiclesRef.where('driverId', isEqualTo: driverId).get();
      return q3.docs.map((d) => Vehicle.fromDocument(d)).toList();
    } catch (e) {
      debugPrint('VehicleService.getDriverVehicles error: $e');
      rethrow;
    }
  }

  /// جلب مركبة بواسطة المعرف
  Future<Vehicle?> getVehicleById(String id) async {
    try {
      final doc = await _vehiclesRef.doc(id).get();

      return Vehicle.fromDocument(doc);
    } catch (e) {
      debugPrint('VehicleService.getVehicleById error: $e');
      rethrow;
    }
  }

    try {
      final data = vehicle.toMap();

      final payload = {

        'maxCapacity': data['maxCapacity'] ?? 0,
        'licenseNumber': data['licenseNumber'] ?? '',
        'isVerified': data['isVerified'] ?? false,
        'notes': data['notes'] ?? '',
        'createdAt': FieldValue.serverTimestamp(),

      debugPrint('VehicleService.createVehicle error: $e');

/// نموذج Agency موحّد وآمن للقراءة والكتابة إلى Firestore.
class Agency {
  final String id;
  final String agencyName;
  final String managerName;
  final dynamic location; // GeoPoint أو Map أو null
  final String mobileNumber;
  final double walletBalance;
  final bool isActive;
  final Timestamp? createdAt;
  final Timestamp? updatedAt;

  Agency({
    required this.id,
    this.agencyName = '',
    this.managerName = '',
    this.location,
    this.mobileNumber = '',
    this.walletBalance = 0.0,
    this.isActive = true,
    this.createdAt,
    this.updatedAt,
  });

  /// قراءة من خريطة (مثلاً من DocumentSnapshot.data())
  factory Agency.fromMap(String id, Map<String, dynamic>? data) {
    final d = data ?? <String, dynamic>{};

    dynamic parseLocation(dynamic v) {
      if (v == null) return null;
      if (v is GeoPoint) return v;
      if (v is Map<String, dynamic>) return v;
      if (v is Map) return Map<String, dynamic>.from(v);
      return v;
    }

    Timestamp? parseTs(dynamic v) {
      if (v == null) return null;
      if (v is Timestamp) return v;
      if (v is DateTime) return Timestamp.fromDate(v);
      if (v is String) {
        final parsed = DateTime.tryParse(v);
        if (parsed != null) return Timestamp.fromDate(parsed);
      }
      if (v is int) {
        try {
          return Timestamp.fromMillisecondsSinceEpoch(v);
        } catch (_) {}
      }
      return null;
    }

    double toDoubleSafe(dynamic v) {
      if (v == null) return 0.0;
      if (v is double) return v;
      if (v is int) return v.toDouble();
      if (v is num) return v.toDouble();
      if (v is String) return double.tryParse(v) ?? 0.0;
      return 0.0;
    }

    return Agency(
      id: id,
      agencyName: (d['agencyName'] ?? '') as String,
      managerName: (d['managerName'] ?? '') as String,
      location: parseLocation(d['location']),
      mobileNumber: (d['mobileNumber'] ?? '') as String,
      walletBalance: toDoubleSafe(d['walletBalance']),
      isActive: d['isActive'] is bool ? d['isActive'] as bool : (d['isActive'] == null ? true : d['isActive'].toString().toLowerCase() == 'true'),
      createdAt: parseTs(d['createdAt']),
      updatedAt: parseTs(d['updatedAt']),
    );
  }

  /// إنشاء من DocumentSnapshot
  factory Agency.fromDoc(DocumentSnapshot doc) {
    final data = doc.data() as Map<String, dynamic>? ?? <String, dynamic>{};
    return Agency.fromMap(doc.id, data);
  }

  /// تحويل إلى Map للحفظ في Firestore
  Map<String, dynamic> toMap() {
    return {
      'agencyName': agencyName,
      'managerName': managerName,
      'location': location,
      'mobileNumber': mobileNumber,
      'walletBalance': walletBalance,
      'isActive': isActive,
      'createdAt': createdAt ?? FieldValue.serverTimestamp(),
      'updatedAt': updatedAt ?? FieldValue.serverTimestamp(),
    }..removeWhere((k, v) => v == null);
  }

  Agency copyWith({
    String? agencyName,
    String? managerName,
    dynamic location,
    String? mobileNumber,
    double? walletBalance,
    bool? isActive,
    Timestamp? createdAt,
    Timestamp? updatedAt,
  }) {

      id: id,
      agencyName: agencyName ?? this.agencyName,

      location: location ?? this.location,
      mobileNumber: mobileNumber ?? this.mobileNumber,
      walletBalance: walletBalance ?? this.walletBalance,
      isActive: isActive ?? this.isActive,
      createdAt: createdAt ?? this.createdAt,
      updatedAt: updatedAt ?? this.updatedAt,
    );
  }

  @override
  String toString() {
    return 'Agency(id: $id, name: $agencyName, manager: $managerName, active: $isActive, wallet: $walletBalance)';
  }

import 'package:cloud_firestore/cloud_firestore.dart';

/// نموذج AgencyTransaction موحّد وآمن للقراءة والكتابة إلى Firestore.
class AgencyTransaction {
  final String id;
  final String agencyId;
  final double amount;
  final String type; // 'credit' or 'debit'
  final String? reason;
  final String? performedBy;
  final Timestamp createdAt;

  AgencyTransaction({
    required this.id,
    required this.agencyId,
    required this.amount,
    required this.type,
    this.reason,
    this.performedBy,
    required this.createdAt,
  });

  /// قراءة من DocumentSnapshot مع تحويل آمن للأنواع
  factory AgencyTransaction.fromDoc(DocumentSnapshot doc) {
    final data = doc.data() as Map<String, dynamic>? ?? <String, dynamic>{};

    double toDoubleSafe(dynamic v) {
      if (v == null) return 0.0;
      if (v is double) return v;
      if (v is int) return v.toDouble();
      if (v is num) return v.toDouble();
      if (v is String) return double.tryParse(v) ?? 0.0;
      return 0.0;
    }

    String s(List<String> keys, [String fallback = '']) {
      for (var k in keys) {
        final v = data[k];
        if (v != null && v.toString().isNotEmpty) return v.toString();
      }
      return fallback;
    }

    Timestamp parseTs(dynamic v) {
      if (v is Timestamp) return v;
      if (v is DateTime) return Timestamp.fromDate(v);
      if (v is String) {
        final parsed = DateTime.tryParse(v);
        if (parsed != null) return Timestamp.fromDate(parsed);
      }
      if (v is int) {
        try {
          return Timestamp.fromMillisecondsSinceEpoch(v);
        } catch (_) {}
      }
      return Timestamp.now();
    }

    return AgencyTransaction(
      id: doc.id,
      agencyId: s(['agencyId', 'agency_id', 'agency']),
      amount: toDoubleSafe(data['amount'] ?? data['value'] ?? 0),
      type: s(['type']),
      reason: (data['reason'] != null && data['reason'].toString().isNotEmpty) ? data['reason'].toString() : null,
      performedBy: (data['performedBy'] != null && data['performedBy'].toString().isNotEmpty) ? data['performedBy'].toString() : null,
      createdAt: data['createdAt'] is Timestamp ? data['createdAt'] as Timestamp : parseTs(data['createdAt']),
    );
  }

  /// تحويل الكائن إلى Map للحفظ في Firestore
  Map<String, dynamic> toMap() {
    return {
      'agencyId': agencyId,
      'amount': amount,
      'type': type,
      'reason': reason,
      'performedBy': performedBy,
      'createdAt': createdAt,
    }..removeWhere((k, v) => v == null);
  }

  AgencyTransaction copyWith({
    String? id,
    String? agencyId,
    double? amount,
    String? type,
    String? reason,
    String? performedBy,
    Timestamp? createdAt,
  }) {
    return AgencyTransaction(
      id: id ?? this.id,
      agencyId: agencyId ?? this.agencyId,
      amount: amount ?? this.amount,
      type: type ?? this.type,
      reason: reason ?? this.reason,
      performedBy: performedBy ?? this.performedBy,
      createdAt: createdAt ?? this.createdAt,
    );
  }

  @override
  String toString() {
    return 'AgencyTransaction(id: $id, agencyId: $agencyId, amount: $amount, type: $type, performedBy: ${performedBy ?? 'N/A'})';
  }
}

  /// إنشاء نموذج من مستند Firestore مع معالجة أنواع مختلفة بأمان
  factory Booking.fromDocument(DocumentSnapshot doc) {
    final data = doc.data() as Map<String, dynamic>? ?? {};

    String s(List<String> keys, [String fallback = '']) {
      for (var k in keys) {
        final v = data[k];
        if (v != null && v.toString().isNotEmpty) return v.toString();
      }
      return fallback;
    }

    int i(List<String> keys, [int fallback = 0]) {
      for (var k in keys) {
        final v = data[k];
        if (v is int) return v;
        if (v is double) return v.toInt();
        if (v is String) {
          final parsed = int.tryParse(v);
          if (parsed != null) return parsed;
        }
      }
      return fallback;
    }

    double d(List<String> keys, [double fallback = 0.0]) {
      for (var k in keys) {
        final v = data[k];
        if (v is num) return v.toDouble();
        if (v is String) {
          final parsed = double.tryParse(v);
          if (parsed != null) return parsed;
        }
      }
      return fallback;
    }

    bool b(List<String> keys, [bool fallback = false]) {
      for (var k in keys) {
        final v = data[k];
        if (v is bool) return v;
        if (v is num) return v != 0;
        if (v is String) {
          final low = v.toLowerCase();
          if (low == 'true' || low == '1') return true;
          if (low == 'false' || low == '0') return false;
        }
      }
      return fallback;
    }

    Timestamp? ts(dynamic v) {
      if (v == null) return null;
      if (v is Timestamp) return v;
      if (v is DateTime) return Timestamp.fromDate(v);
      if (v is String) {
        final parsed = DateTime.tryParse(v);
        if (parsed != null) return Timestamp.fromDate(parsed);
      }
      if (v is int) {
        try {
          return Timestamp.fromMillisecondsSinceEpoch(v);
        } catch (_) {}
      }
      return null;
    }

    List<int>? parseSeatIndexes(dynamic raw) {
      if (raw == null) return null;
      if (raw is List) {
        try {
          return raw.map<int>((e) {
            if (e is int) return e;
            if (e is double) return e.toInt();
            if (e is String) return int.tryParse(e) ?? -1;
            return -1;
          }).where((v) => v >= 0).toList();
        } catch (_) {
          return null;
        }
      }
      // إذا كان مخزناً كقيمة مفردة (legacy) مثل 0 أو "0"
      if (raw is int) return [raw];
      if (raw is double) return [raw.toInt()];
      if (raw is String) {
        final parsed = int.tryParse(raw);
        if (parsed != null) return [parsed];
      }
      return null;
    }

    // قراءة reservedUntil قد تكون نص أو Timestamp
    final reservedRaw = data['reservedUntil'] ?? data['reserved_until'];
    final reservedTimestamp = ts(reservedRaw);

    return Booking(
      id: doc.id,
      tripId: s(['tripId', 'trip_id']),
      passengerId: s(['passengerId', 'passenger_id']),
      passengerNickname: s(['passengerNickname', 'passenger_nickname', 'passengerName']),
      driverId: s(['driverId', 'driver_id']),
      amountPaid: d(['amountPaid', 'amount_paid', 'paidAmount']),
      totalPrice: d(['totalPrice', 'total_price', 'price']),
      bookedSeatsCount: i(['bookedSeatsCount', 'seatsCount', 'seats_count'], 1),
      isFamilyBooking: b(['isFamilyBooking', 'is_family_booking']),
      childrenCount: i(['childrenCount', 'children_count'], 0),
      bookingStatus: s(['bookingStatus', 'status'], 'pending'),
      paymentStatus: s(['paymentStatus', 'payment_status'], 'unpaid'),
      paymentMethod: s(['paymentMethod', 'payment_method']),
      paymentIntentId: (data['paymentIntentId'] ?? data['payment_intent_id'])?.toString(),
      transactionId: s(['transactionId', 'transaction_id']),
      paidAt: ts(data['paidAt'] ?? data['paid_at']),
      seatIndexes: parseSeatIndexes(data['seatIndexes'] ?? data['seat_indexes']),
      seatNumber: (data['seatNumber'] ?? data['seat_number']) is num
          ? (data['seatNumber'] ?? data['seat_number']).toInt()
          : null,
      reservedUntil: reservedTimestamp,
      notes: data['notes']?.toString(),
      canceledAt: ts(data['canceledAt'] ?? data['canceled_at']),
      createdAt: ts(data['createdAt'] ?? data['created_at']),
      updatedAt: ts(data['updatedAt'] ?? data['updated_at']),
      bookingService: s(['BookingService', 'bookingService']),
      pdfService: s(['PdfService', 'pdfService']),
    );
  }

/// تمثيل شريحة مسافة لرسوم الحكومة
class DistanceTier {
  final double minKm;
  final double maxKm;
  final double amount;

  DistanceTier({
    required this.minKm,
    required this.maxKm,
    required this.amount,
  });

  factory DistanceTier.fromMap(Map<String, dynamic> m) {
    double _toDouble(dynamic v, [double fallback = 0.0]) {
      if (v == null) return fallback;
      if (v is num) return v.toDouble();
      if (v is String) return double.tryParse(v) ?? fallback;
      return fallback;
    }

    return DistanceTier(
      minKm: _toDouble(m['minKm'] ?? m['min_km'] ?? m['min'] ?? 0.0),
      maxKm: _toDouble(m['maxKm'] ?? m['max_km'] ?? m['max'] ?? double.infinity),
      amount: _toDouble(m['amount'] ?? m['fee'] ?? 0.0),
    );
  }

  Map<String, dynamic> toMap() => {
    'minKm': minKm,
    'maxKm': maxKm,
    'amount': amount,
  };
}

/// نموذج قاعدة رسوم حكومية (gov fee rule)
class GovFeeRule {
  final String id;
  final String origin; // اسم نقطة الانطلاق أو '*'
  final String destination; // اسم الوجهة أو '*'
  final bool exact; // تطابق دقيق origin/destination
  final bool distanceBased; // هل تعتمد الرسوم على المسافة
  final double fixedAmount; // مبلغ ثابت إن وُجد
  final List<DistanceTier> distanceTiers; // شرائح المسافة
  final int priority; // أولوية القاعدة
  final bool isActive;
  final Timestamp createdAt;
  final Timestamp? updatedAt;

  GovFeeRule({
    required this.id,
    required this.origin,
    required this.destination,
    required this.exact,
    required this.distanceBased,
    required this.fixedAmount,
    required this.distanceTiers,
    required this.priority,
    required this.isActive,
    required this.createdAt,
    this.updatedAt,
  });

  /// مساعدة لتحويل قيمة إلى double بأمان
  static double _toDoubleSafe(dynamic v, [double fallback = 0.0]) {
    if (v == null) return fallback;
    if (v is double) return v;
    if (v is int) return v.toDouble();
    if (v is num) return v.toDouble();
    if (v is String) return double.tryParse(v) ?? fallback;
    return fallback;
  }

  /// مساعدة لتحويل قيمة إلى int بأمان
  static int _toIntSafe(dynamic v, [int fallback = 0]) {
    if (v == null) return fallback;
    if (v is int) return v;
    if (v is double) return v.toInt();
    if (v is String) return int.tryParse(v) ?? fallback;
    return fallback;
  }

  /// قراءة Timestamp بأمان من أنواع متعددة
  static Timestamp _parseTimestamp(dynamic v) {
    if (v is Timestamp) return v;
    if (v is DateTime) return Timestamp.fromDate(v);
    if (v is String) {
      final parsed = DateTime.tryParse(v);
      if (parsed != null) return Timestamp.fromDate(parsed);
    }
    if (v is int) {
      try {
        return Timestamp.fromMillisecondsSinceEpoch(v);
      } catch (_) {}
    }
    return Timestamp.now();
  }

  /// إنشاء من مستند Firestore
  factory GovFeeRule.fromDoc(DocumentSnapshot doc) {
    final data = doc.data() as Map<String, dynamic>? ?? {};

    // distanceTiers قد تكون Map أو List أو JSON string

    List<DistanceTier> parseTiers(dynamic raw) {
      if (raw == null) return [];

      // إذا كانت خريطة (map) من كائنات
      if (raw is Map) {
        try {
          return raw.values.map((e) {
            final m = e is Map<String, dynamic> ? e : Map<String, dynamic>.from(e as Map);
            return DistanceTier.fromMap(m);
          }).toList();
        } catch (_) {
          return [];
        }
      }

      // إذا كانت قائمة
      if (raw is List) {
        return raw.map((e) {
          if (e is Map<String, dynamic>) return DistanceTier.fromMap(e);
          if (e is Map) return DistanceTier.fromMap(Map<String, dynamic>.from(e));
          return DistanceTier(minKm: 0.0, maxKm: double.infinity, amount: 0.0);
        }).toList();
      }

      // إذا كانت نص JSON
      if (raw is String) {
        final parsed = raw.trim();
        if (parsed.isEmpty) return [];
        try {
          final decoded = jsonDecode(parsed);
          if (decoded is List) {
            return decoded.map((e) => DistanceTier.fromMap(Map<String, dynamic>.from(e))).toList();
          }
          if (decoded is Map) {
            return decoded.values.map((e) => DistanceTier.fromMap(Map<String, dynamic>.from(e))).toList();
          }
        } catch (_) {
          // JSON غير صالح — نرجع قائمة فارغة بدل رمي استثناء
          return [];
        }
      }

      // أي نوع آخر
      return [];
    }
    final tiers = parseTiers(data['distanceTiers'] ?? data['distance_tiers'] ?? data['tiers']);

    return GovFeeRule(
      id: doc.id,
      origin: (data['origin'] ?? '*').toString(),
      destination: (data['destination'] ?? '*').toString(),
      exact: (data['exact'] ?? false) as bool,
      distanceBased: (data['distanceBased'] ?? data['distance_based'] ?? false) as bool,
      fixedAmount: _toDoubleSafe(data['fixedAmount'] ?? data['fixed_amount'] ?? 0.0),
      distanceTiers: tiers,
      priority: _toIntSafe(data['priority'] ?? 0),
      isActive: data['isActive'] is bool ? data['isActive'] as bool : (data['isActive']?.toString().toLowerCase() == 'true'),
      createdAt: data['createdAt'] is Timestamp ? data['createdAt'] as Timestamp : _parseTimestamp(data['createdAt']),
      updatedAt: data['updatedAt'] is Timestamp ? data['updatedAt'] as Timestamp : (data['updatedAt'] != null ? _parseTimestamp(data['updatedAt']) : null),
    );
  }

  /// تحويل إلى Map للحفظ في Firestore
  Map<String, dynamic> toMap() {
    return {
      'origin': origin,
      'destination': destination,
      'exact': exact,
      'distanceBased': distanceBased,
      'fixedAmount': fixedAmount,
      'distanceTiers': distanceTiers.map((t) => t.toMap()).toList(),
      'priority': priority,
      'isActive': isActive,
      'createdAt': createdAt,
      'updatedAt': updatedAt ?? FieldValue.serverTimestamp(),
    }..removeWhere((k, v) => v == null);
  }

  /// احسب الرسوم الحكومية لرحلة بناءً على المسافة والقاعدة
  /// ترجع null إذا لم تنطبق القاعدة (مثلاً origin/destination لا تطابق)
  double? calculateFee({required double distanceKm, required String origin, required String destination}) {
    // تحقق من التطابق مع origin/destination
    if (exact) {
      if (this.origin != '*' && this.origin != origin) return null;
      if (this.destination != '*' && this.destination != destination) return null;
    } else {
      if (this.origin != '*' && this.origin != origin) return null;
      if (this.destination != '*' && this.destination != destination) return null;
    }

    if (!isActive) return null;

    if (!distanceBased) {
      return fixedAmount;
    }

    // البحث في الشرائح
    for (final tier in distanceTiers) {
      final minOk = distanceKm >= tier.minKm;
      final maxOk = tier.maxKm.isFinite ? distanceKm <= tier.maxKm : true;
      if (minOk && maxOk) return tier.amount;
    }

    // إذا لم يطابق أي شريحة، استخدم fixedAmount إن وُجد
    return fixedAmount;
  }
}

/// نموذج إشعار التطبيق (AppNotification)
/// يدعم حقول إضافية: actionUrl, priority, sendNotification, transactionId
class AppNotification {
  final String id;
  final String userId;
  final String title;
  final String body;
  final String type; // مثال: 'booking', 'payment', 'system', ...
  final Map<String, dynamic>? metadata; // بيانات إضافية قابلة للتوسع
  final bool isRead;

  final String? transactionId; // معرف عملية إن وُجد
  final Timestamp? createdAt;
  final Timestamp? updatedAt;

  AppNotification({
    required this.id,
    required this.userId,
    required this.title,
    required this.body,
    this.type = 'general',
    this.metadata,
    this.isRead = false,
    this.actionUrl,
    this.priority = 'normal',
    this.sendNotification = true,
    this.transactionId,
    this.createdAt,
    this.updatedAt,

  /// تحويل الكائن إلى Map لحفظه في Firestore
  Map<String, dynamic> toMap() {

      'userId': userId,
      'title': title,
      'body': body,
      'type': type,
      'metadata': metadata,
      'isRead': isRead,
      'actionUrl': actionUrl,
      'priority': priority,
      'sendNotification': sendNotification,
      'transactionId': transactionId,
      'createdAt': createdAt ?? FieldValue.serverTimestamp(),
      'updatedAt': updatedAt ?? FieldValue.serverTimestamp(),
    }..removeWhere((key, value) => value == null);
  }

  /// مساعدة لتحويل قيمة قد تكون JSON نصي إلى Map
  static Map<String, dynamic>? _tryParseMetadata(dynamic v) {
    if (v == null) return null;
    if (v is Map<String, dynamic>) return v;
    if (v is Map) return Map<String, dynamic>.from(v);
    if (v is String) {
      final s = v.trim();
      if (s.isEmpty) return null;
      try {
        final decoded = json.decode(s);
        if (decoded is Map<String, dynamic>) return decoded;
        if (decoded is Map) return Map<String, dynamic>.from(decoded);
      } catch (_) {
        // ليس JSON صالحًا — تجاهل
      }
    }
    return null;
  }

  /// إنشاء نموذج من مستند Firestore مع دعم camelCase و snake_case
  factory AppNotification.fromDocument(DocumentSnapshot doc) {
    final data = doc.data() as Map<String, dynamic>? ?? {};

    String s(List<String> keys, [String fallback = '']) {
      for (var k in keys) {
        final v = data[k];
        if (v != null && v.toString().isNotEmpty) return v.toString();
      }
      return fallback;
    }

    bool b(List<String> keys, [bool fallback = false]) {
      for (var k in keys) {
        final v = data[k];
        if (v is bool) return v;
        if (v is num) return v != 0;
        if (v is String) {
          final low = v.toLowerCase();
          if (low == 'true' || low == '1') return true;
          if (low == 'false' || low == '0') return false;
        }
      }
      return fallback;
    }

    Map<String, dynamic>? m(List<String> keys) {
      for (var k in keys) {
        final v = data[k];
        final parsed = _tryParseMetadata(v);
        if (parsed != null) return parsed;
      }
      return null;
    }

    Timestamp? ts(List<String> keys) {
      for (var k in keys) {
        final v = data[k];
        if (v is Timestamp) return v;
        if (v is String) {
          final parsed = DateTime.tryParse(v);

    final data = doc.data() as Map<String, dynamic>? ?? {};

    String s(List<String> keys, [String fallback = '']) {
      for (var k in keys) {
        final v = data[k];
        if (v != null && v.toString().isNotEmpty) return v.toString();

class Trip {

  Trip({
    required this.id,
    this.title = '',
    this.description = '',
    this.driverId = '',
    this.driverName = '',
    this.agencyId,
    this.tripCategory = '',
    this.origin = '',
    this.destination = '',
    this.vehicleType = '',
    this.vehicleId,
    this.totalSeats = 0,
    this.availableSeats = 0,
    List<bool>? seatStatus,
    this.departureTime,
    this.date,
    this.time = '',
    this.flexibleWindow = '',
    this.status = 'active',
    this.price = 0.0,
    this.driverRating = 0.0,

    this.carOwnerOfficialName = '',
    this.govFeeAmount,
    this.distanceKm,
    this.manifestUrl,
    this.manifestStoragePath,
    this.manifestGeneratedAt,
    this.manifestStatus,
    this.cancellationPolicy,
    this.createdAt,
    this.updatedAt,
  }) : seatStatus = seatStatus ?? List<bool>.filled(totalSeats > 0 ? totalSeats : 0, true);

  // =========================
  // Factory: from Firestore doc
  // =========================
  factory Trip.fromDocument(DocumentSnapshot doc) {
    final data = doc.data() as Map<String, dynamic>? ?? {};

    int _parseInt(dynamic v) {
      if (v is int) return v;
      if (v is double) return v.toInt();
      if (v is String) return int.tryParse(v) ?? 0;
      return 0;
    }

    double _parseDouble(dynamic v) {
      if (v is num) return v.toDouble();
      if (v is String) return double.tryParse(v) ?? 0.0;
      return 0.0;
    }

    DateTime? _parseDate(dynamic v) {
      if (v is Timestamp) return v.toDate();
      if (v is DateTime) return v;
      if (v is String) return DateTime.tryParse(v);
      if (v is int) return DateTime.fromMillisecondsSinceEpoch(v);
      return null;
    }

    List<bool> _parseSeatStatus(dynamic raw, int total) {
      if (raw is List) {
        final parsed = raw.map<bool>((e) {
          if (e is bool) return e;
          if (e is num) return e != 0;
          if (e is String) return e.toLowerCase() == 'true' || e == '1';
          return false;
        }).toList();
        if (total > 0 && parsed.length != total) {
          return List<bool>.filled(total, true);
        }
        return parsed;
      }
      return List<bool>.filled(total, true);
    }

    final totalSeats = _parseInt(data['totalSeats']);

      id: doc.id,

      date: _parseDate(data['date']),
      time: data['time'] ?? '',
      flexibleWindow: data['flexibleWindow'] ?? '',
      status: data['status'] ?? 'active',
      price: _parseDouble(data['price']),
      driverRating: _parseDouble(data['driverRating']),
      carPlateNumber: data['carPlateNumber'] ?? data['plateNumber'] ?? '',
      carOwnerOfficialName: data['carOwnerOfficialName'] ?? '',
      govFeeAmount: data['govFeeAmount'] != null ? _parseDouble(data['govFeeAmount']) : null,
      distanceKm: data['distanceKm'] != null ? _parseDouble(data['distanceKm']) : null,
      manifestUrl: data['manifestUrl'],
      manifestStoragePath: data['manifestStoragePath'],
      manifestGeneratedAt: data['manifestGeneratedAt'] is Timestamp ? data['manifestGeneratedAt'] : null,
      manifestStatus: data['manifestStatus'],
      cancellationPolicy: data['cancellationPolicy'],
      createdAt: data['createdAt'] is Timestamp ? data['createdAt'] : null,
      updatedAt: data['updatedAt'] is Timestamp ? data['updatedAt'] : null,
    );
  }

  // =========================
  // Convert to Map for Firestore
  // =========================
  Map<String, dynamic> toMap() {

import 'dart:convert';
import 'package:cloud_firestore/cloud_firestore.dart';

/// نموذج Vehicle موحّد وآمن للقراءة والكتابة إلى Firestore.
/// يدعم أسماء الحقول البديلة (camelCase / snake_case) ويتعامل مع أنواع متعددة بأمان.
class Vehicle {
  final String id;
  final String plateNumber;
  final String vehicleOwner; // userId الموحد الحالي
  final String? ownerId;     // حقل احتياطي في المستندات القديمة
  final String? driverId;    // مرجع اختياري للسائق المرتبط بالمركبة
  final String? driverName;
  final String vehicleType;  // مثال: 'sedan', 'van', 'motorbike'
  final String makeModelYear;
  final String color;
  final int maxCapacity;     // أقصى عدد ركاب
  final String licenseNumber;
  final bool isVerified;
  final bool sGovernmentRegistered;
  final String? notes;
  final Map<String, dynamic>? extra;
  final List<String> vehicleDocuments;
  final Timestamp? insuranceExpiryDate;
  final String? lastInspectionDate;
  final String? carOwnerOfficialName;
  final Timestamp? createdAt;
  final Timestamp? updatedAt;

  Vehicle({
    required this.id,
    this.plateNumber = '',
    this.vehicleOwner = '',
    this.ownerId,
    this.driverId,
    this.driverName,
    this.vehicleType = '',
    this.makeModelYear = '',
    this.color = '',
    this.maxCapacity = 0,
    this.licenseNumber = '',
    this.isVerified = false,
    this.sGovernmentRegistered = false,
    this.notes,
    this.extra,
    this.vehicleDocuments = const [],
    this.insuranceExpiryDate,
    this.lastInspectionDate,
    this.carOwnerOfficialName,
    this.createdAt,
    this.updatedAt,
  });

  /// تحويل الكائن إلى Map للحفظ في Firestore
  Map<String, dynamic> toMap() {
    final map = <String, dynamic>{
      'plateNumber': plateNumber,
      'vehicleOwner': vehicleOwner,
      'ownerId': ownerId,
      'driverId': driverId,
      'driverName': driverName,
      'vehicleType': vehicleType,
      'makeModelYear': makeModelYear,
      'color': color,
      'maxCapacity': maxCapacity,
      'licenseNumber': licenseNumber,
      'isVerified': isVerified,
      'sGovernmentRegistered': sGovernmentRegistered,
      'notes': notes,
      'extra': extra,
      'vehicleDocuments': vehicleDocuments,
      'insuranceExpiryDate': insuranceExpiryDate,
      'lastInspectionDate': lastInspectionDate,
      'carOwnerOfficialName': carOwnerOfficialName,
      'createdAt': createdAt ?? FieldValue.serverTimestamp(),
      'updatedAt': updatedAt ?? FieldValue.serverTimestamp(),
    };
    map.removeWhere((k, v) => v == null);
    return map;
  }

  /// مساعدة لتحويل قيمة قد تكون JSON نصي إلى Map
  static Map<String, dynamic>? _tryParseExtra(dynamic v) {
    if (v == null) return null;
    if (v is Map<String, dynamic>) return v;
    if (v is Map) return Map<String, dynamic>.from(v);
    if (v is String) {
      final s = v.trim();
      if (s.isEmpty) return null;
      try {
        final decoded = json.decode(s);
        if (decoded is Map<String, dynamic>) return decoded;
        if (decoded is Map) return Map<String, dynamic>.from(decoded);
      } catch (_) {}
    }
    return null;
  }

  /// إنشاء نموذج من مستند Firestore مع دعم camelCase و snake_case وأنواع متعددة
  factory Vehicle.fromDocument(DocumentSnapshot doc) {
    final data = doc.data() as Map<String, dynamic>? ?? {};

    String s(List<String> keys, [String fallback = '']) {
      for (var k in keys) {
        final v = data[k];
        if (v != null && v.toString().isNotEmpty) return v.toString();
      }
      return fallback;
    }

    int i(List<String> keys, [int fallback = 0]) {
      for (var k in keys) {
        final v = data[k];
        if (v is int) return v;
        if (v is double) return v.toInt();
        if (v is String) return int.tryParse(v) ?? fallback;
      }
      return fallback;
    }

    bool b(List<String> keys, [bool fallback = false]) {
      for (var k in keys) {
        final v = data[k];
        if (v is bool) return v;
        if (v is num) return v != 0;
        if (v is String) {
          final low = v.toLowerCase();
          if (low == 'true' || low == '1') return true;
          if (low == 'false' || low == '0') return false;
        }
      }
      return fallback;
    }

    Timestamp? ts(List<String> keys) {
      for (var k in keys) {
        final v = data[k];
        if (v is Timestamp) return v;
        if (v is String) {
          final parsed = DateTime.tryParse(v);
          if (parsed != null) return Timestamp.fromDate(parsed);
        }
        if (v is int) {
          try {
            return Timestamp.fromMillisecondsSinceEpoch(v);
          } catch (_) {}
        }
      }
      return null;
    }

    List<String> list(List<String> keys) {
      for (var k in keys) {
        final v = data[k];
        if (v is List) {
          return v.map((e) => e?.toString() ?? '').where((e) => e.isNotEmpty).toList();
        }
      }
      return [];
    }

    Map<String, dynamic>? map(List<String> keys) {
      for (var k in keys) {
        final v = data[k];
        final parsed = _tryParseExtra(v);
        if (parsed != null) return parsed;
      }
      return null;
    }

    return Vehicle(
      id: doc.id,
      plateNumber: s(['plateNumber', 'plate_number', 'plate']),
      vehicleOwner: s(['vehicleOwner', 'vehicle_owner', 'ownerUserId', 'owner']),
      ownerId: s(['ownerId', 'owner']),
      driverId: s(['driverId', 'driver_id']),
      driverName: s(['driverName']),
      vehicleType: s(['vehicleType', 'vehicle_type', 'type']),
      makeModelYear: s(['makeModelYear', 'make_model_year', 'makeModel', 'modelYear']),
      color: s(['color']),
      maxCapacity: i(['maxCapacity', 'max_capacity', 'capacity']),
      licenseNumber: s(['licenseNumber', 'license_number', 'registrationNumber']),
      isVerified: b(['isVerified', 'is_verified', 'verified']),
      sGovernmentRegistered: b(['sGovernmentRegistered', 'governmentRegistered']),
      notes: s(['notes', 'note', 'description']).isNotEmpty ? s(['notes', 'note', 'description']) : null,
      extra: map(['extra', 'meta', 'metadata']),
      vehicleDocuments: list(['vehicleDocuments']),
      insuranceExpiryDate: ts(['insuranceExpiryDate']),
      lastInspectionDate: s(['lastInspectionDate']),
      carOwnerOfficialName: s(['carOwnerOfficialName']),
      createdAt: ts(['createdAt', 'created_at', 'created']),
      updatedAt: ts(['updatedAt', 'updated_at']),
    );
  }
